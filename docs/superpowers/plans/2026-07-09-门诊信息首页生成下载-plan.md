# 门诊信息首页生成下载 — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build an HISUI page that lets users select year-month, generate CSV files from the OEPInfoPageInterface.GetOutpatientList query to /tmp/excel/, list existing files, and download them.

**Architecture:** One backend class (`web.YZSY.DHCMRInfoPageDownLoad`) with 3 ClassMethods, one CSP page with external JS. Frontend calls backend via `$cm()`/`$m()`, file download uses Blob + createObjectURL.

**Tech Stack:** ObjectScript CLS, CSP, HISUI (jQuery + easyUI), vanilla JS (IE11-compatible)

---

### File Structure

| File | Responsibility |
|------|---------------|
| `web.YZSY.DHCMRInfoPageDownLoad.cls` | Generate CSV, list files, read file for download |
| `dhcmrinfopagedownload.csp` | Page layout, HISUI components, references JS |
| `dhcmrinfopagedownload.js` | All frontend logic: load list, generate, download |

### Interfaces

```
CSP ──$cm({MethodName:"GenerateCSV", YearMonth:"2026-06"})──▶ cls.GenerateCSV()
CSP ──$cm({MethodName:"ListFiles"})──────────────────────▶ cls.ListFiles()
CSP ──$m({MethodName:"DownloadFile", fileName:"xxx.csv"})──▶ cls.DownloadFile()
```

```objectscript
Class web.YZSY.DHCMRInfoPageDownLoad Extends DHCDoc.Util.RegisteredObject
{

/// 固定参数
Parameter QUERYCLASS = "EMRservice.HISInterface.OEPInfoPageInterface";
Parameter QUERYNAME = "GetOutpatientList";
Parameter OUTPUTDIR = "/tmp/excel/";
Parameter FILEPREFIX = "门诊信息首页_";

/// ExcelTitle — 字段码表头
Parameter EXCELTITLE = "MA50^MA11^MA12C^MA13^MA21C^MA15C^MA19C^MA20N^MA20^MA26^MA35^MA51C^MA52^MA53C^MA54^MA02^MB39^MB41^MB42^MB46^MB47C^MB49N^MB50C^MB44C^MB43C^MB54C^MB45C^MB51C^MB52C^MB53^MC51^MC01C^MC02N^MC52x01C^MC52x01N^MC52x02C^MC52x02N^MC52x03C^MC52x03N^MC52x04C^MC52x04N^MC52x05C^MC52x05N^MC52x06C^MC52x06N^MC52x07C^MC52x07N^MC52x08C^MC52x08N^MC52x09C^MC52x09N^MC52x10C^MC52x10N^MC16x01^MC16x02^MC16x03^MC16x04^MC16x05^MC15x01N^MC15x02N^MC15x03N^MC15x04N^MC15x05N^MC14x01C^MC14x02C^MC14x03C^MC14x04C^MC14x05C^MC18x01N^MC18x02N^MC18x03N^MC18x04N^MC18x05N^MC22x01C^MC22x02C^MC22x03C^MC22x04C^MC22x05C^MC23x01^MC23x02^MC23x03^MC23x04^MC23x05^MC17x01^MC17x02^MC17x03^MC17x04^MC17x05^MD01^MD09^MD11^MD12^MD13^MD14^MD15^MD16^MD17^MD18^MD19^MD19x01^MD20^MD20x01^MD20x02^MD21^MD22^MD23^MD23x01^MD24^MD25^MD26^MD27^MD28^MD29^MD30^MD31^MD32^MD33^MD34";

/// 根据年月生成CSV文件
/// 入参 YearMonth: "2026-06" 格式
/// 返回 JSON: {"success":true,"fileName":"门诊信息首页_2026-06.csv","msg":""}
ClassMethod GenerateCSV(YearMonth As %String) As %String
{
	s tmpspl = $c(94)

	s OutDir = ..#OUTPUTDIR
	s FilePrefix = ..#FILEPREFIX
	s ExcelTitle = ..#EXCELTITLE

	; 解析年月
	s year = $p(YearMonth, "-", 1)
	s month = $p(YearMonth, "-", 2)
	i (year = "") || (month = "") q "{""success"":false,""msg"":""年月格式错误，请传 YYYY-MM""}"

	; 计算月初和月末日期
	s SDate = year _ "-" _ month _ "-01"
	s SDateH = ..%ZDH(SDate)
	s nextMonthH = $s(month = 12:..%ZDH((year + 1) _ "-01-01"), 1:..%ZDH(year _ "-" _ (month + 1) _ "-01"))
	s EDateH = nextMonthH - 1
	s EDate = ..%ZD(EDateH)

	; 生成文件名
	s CurrDate = $tr($zd(+$h, 3), "-", "")
	s CurrTime = $tr($zt($p($h, ",", 2), 1), ":", "")
	s FileName = FilePrefix _ YearMonth _ ".csv"

	; 确保输出目录存在
	i '##class(%File).DirectoryExists(OutDir) {
		s dirOk = ##class(%File).CreateDirectoryChain(OutDir)
		i 'dirOk q "{""success"":false,""msg"":""无法创建输出目录 "_OutDir _ """}"
	}

	s FullPath = OutDir _ FileName

	; 打开文件
	s File = ##class(%File).%New(FullPath)
	d File.Open("NWRS")

	; 执行 Query
	s ClassName = ..#QUERYCLASS
	s QueryName = ..#QUERYNAME
	s rs = ##class(%ResultSet).%New(ClassName _ ":" _ QueryName)
	s sc = rs.Execute(SDate, EDate)
	i $$$ISERR(sc) {
		d File.Close()
		q "{""success"":false,""msg"":""Query 执行失败: "_SDate_" ~ "_EDate_"""}"
	}
	s Columns = rs.GetColumnCount()

	; 写入表头 (tab 分隔)
	d File.WriteLine($tr(ExcelTitle, tmpspl, $c(9)))

	; 写入数据行
	f {
		s sc = rs.Next(.sc)
		q:sc = 0
		i $$$ISOK(sc) {
			s rowData = ""
			f colIdx = 1:1:Columns {
				i rowData = "" {
					s rowData = rs.GetData(colIdx)
				} else {
					s rowData = rowData _ $c(9) _ rs.GetData(colIdx)
				}
			}
			d File.WriteLine(rowData)
		} else {
			q
		}
	}
	i $$$ISERR(sc) d File.Close() q "{""success"":false,""msg"":""数据读取异常""}"

	d File.Flush()
	d File.%Save()
	d File.Close()

	q "{""success"":true,""fileName"":""" _ FileName _ """}"
}

/// 列出 /tmp/excel/ 下的门诊信息首页 CSV 文件
/// 返回 JSON 数组: [{"name":"门诊信息首页_2026-06.csv","size":"12.5 KB","date":"2026-07-09 08:31:43"}]
ClassMethod ListFiles() As %String
{
	s OutDir = ..#OUTPUTDIR
	s FilePrefix = ..#FILEPREFIX

	i '##class(%File).DirectoryExists(OutDir) {
		q "[]"
	}

	s json = "["
	s first = 1

	; 遍历目录
	s rs = ##class(%ResultSet).%New("%File:FileSet")
	s sc = rs.Execute(OutDir, FilePrefix _ "*.csv", "Name", 1)
	i $$$ISERR(sc) q "[]"

	f {
		s ok = rs.Next(.sc)
		q:ok = 0
		i '$$$ISOK(sc) q

		s name = rs.Get("Name")
		s fullPath = OutDir _ name

		; 文件大小 (KB)
		s sizeBytes = ##class(%File).GetFileSize(fullPath)
		s sizeKB = sizeBytes \ 1024
		i sizeKB >= 1024 {
			s sizeStr = (sizeKB \ 1024) _ "." _ (sizeKB # 1024 * 100 \ 1024) _ " MB"
		} else {
			s sizeStr = sizeKB _ " KB"
		}

		; 修改时间
		s dateModified = ##class(%File).GetFileDateModified(fullPath, 1)
		s dateStr = $s(dateModified '= "": $zdt(dateModified, 3, 1), 1: "")

		s:'first json = json _ ","
		s first = 0
		s json = json _ "{""name"":""" _ name _ """,""size"":""" _ sizeStr _ """,""date"":""" _ dateStr _ """}"
	}

	q json _ "]"
}

/// 下载文件 — 返回文件原始内容（前端通过 Blob 下载）
/// 返回文件文本内容（CSV 为文本文件）
ClassMethod DownloadFile(fileName As %String) As %String
{
	s OutDir = ..#OUTPUTDIR
	s fullPath = OutDir _ fileName

	; 安全检查：防止路径穿越
	i fileName [ "/" q ""
	i fileName [ "\" q ""
	i fileName [ ".." q ""
	i '##class(%File).Exists(fullPath) q ""

	s file = ##class(%File).%New(fullPath)
	s sc = file.Open("R")
	i $$$ISERR(sc) q ""

	s content = ""
	f {
		s line = file.ReadLine(.sc)
		q:sc = -1
		s content = content _ line _ $c(13, 10)
	}

	d file.Close()

	q content
}

}
```

```javascript
// 门诊信息首页生成下载 — 前端 JS
// IE11 兼容：全部 var + function，无 ES6+

var CLASS_NAME = "web.YZSY.DHCMRInfoPageDownLoad";

// ========== 初始化 ==========
$(function () {
    initDateSelectors();
    initDatagrid();
    loadFileList();
});

// 初始化年月选择器
function initDateSelectors() {
    var now = new Date();
    var curYear = now.getFullYear();
    var curMonth = now.getMonth() + 1;

    // 年份下拉（最近5年）
    var yearData = [];
    for (var y = curYear; y >= curYear - 4; y--) {
        yearData.push({ value: y.toString(), text: y + "年" }); // 年
    }
    $("#selYear").combobox({
        data: yearData,
        valueField: "value",
        textField: "text",
        panelHeight: "auto",
        editable: false,
        onLoadSuccess: function () {
            $(this).combobox("setValue", curYear.toString());
        }
    });

    // 月份下拉
    var monthData = [];
    for (var m = 1; m <= 12; m++) {
        monthData.push({ value: padZero(m), text: m + "月" }); // 月
    }
    $("#selMonth").combobox({
        data: monthData,
        valueField: "value",
        textField: "text",
        panelHeight: "auto",
        editable: false,
        onLoadSuccess: function () {
            $(this).combobox("setValue", padZero(curMonth));
        }
    });
}

// 初始化文件列表 datagrid
function initDatagrid() {
    $("#dgFiles").datagrid({
        fit: true,
        border: false,
        rownumbers: true,
        singleSelect: true,
        pagination: false,
        columns: [[
            { field: "name", title: "文件名", width: 350 },      // 文件名
            { field: "size", title: "大小", width: 100, align: "center" }, // 大小
            { field: "date", title: "生成时间", width: 180, align: "center" }, // 生成时间
            { field: "op", title: "操作", width: 100, align: "center",
              formatter: function (val, row) {
                  return '<a class="hisui-linkbutton" onclick="downloadFile(\'' + row.name + '\')" data-options="iconCls:\'icon-w-download\'">下载</a>'; // 下载
              }
            }
        ]]
    });
}

// 获取当前选中的年月值 "YYYY-MM"
function getSelectedYM() {
    var year = $("#selYear").combobox("getValue");
    var month = $("#selMonth").combobox("getValue");
    if (!year || !month) {
        return "";
    }
    return year + "-" + month;
}

// 年月变更回调（预留）
function onYearMonthChange() {
    // 切换年月时自动刷新列表
}

// 加载文件列表
function loadFileList() {
    $cm({
        ClassName: CLASS_NAME,
        MethodName: "ListFiles"
    }, function (data) {
        $("#dgFiles").datagrid("loadData", data || []);
    });
}

// 生成 CSV
function generateCSV() {
    var ym = getSelectedYM();
    if (!ym) {
        $.messager.alert("提示", "请先选择年月"); // 提示: 请先选择年月
        return;
    }

    var btn = $("#btnGenerate");
    btn.linkbutton("disable");

    $.messager.progress({ title: "请稍候", msg: "正在生成 CSV 文件..." }); // 请稍候: 正在生成CSV文件...

    $cm({
        ClassName: CLASS_NAME,
        MethodName: "GenerateCSV",
        YearMonth: ym
    }, function (rs) {
        $.messager.progress("close");
        btn.linkbutton("enable");

        if (rs.success) {
            $.messager.alert("提示", "生成成功！\n文件：" + rs.fileName); // 提示: 生成成功
            loadFileList();
        } else {
            $.messager.alert("错误", "生成失败：" + (rs.msg || "未知错误")); // 错误: 生成失败
        }
    });
}

// 下载文件
function downloadFile(fileName) {
    $.messager.progress({ title: "请稍候", msg: "正在下载..." }); // 请稍候: 正在下载...

    $m({
        ClassName: CLASS_NAME,
        MethodName: "DownloadFile",
        fileName: fileName
    }, function (content) {
        $.messager.progress("close");

        if (!content) {
            $.messager.alert("错误", "文件不存在或为空"); // 错误: 文件不存在或为空
            return;
        }

        // 创建 BOM 的 CSV Blob 并触发下载
        var bom = "﻿";
        var blob = new Blob([bom + content], { type: "text/csv;charset=utf-8" });
        var url = URL.createObjectURL(blob);
        var a = document.createElement("a");
        a.href = url;
        a.download = fileName;
        document.body.appendChild(a);
        a.click();
        document.body.removeChild(a);
        URL.revokeObjectURL(url);
    });
}

// 个位数补零
function padZero(n) {
    return n < 10 ? "0" + n : "" + n;
}
```
