## 1. 动态字段选择器

**场景**：页面需要从大量字段（几十到几百个）中选择输出列，支持分组浏览、关键词搜索、拖拽排序。

### 输入

```javascript
// FIELD_META: 字段元数据数组，来自后端 GetFieldMeta()
// [{field, title, type, piece, group}, ...]

// FIELD_GROUPS: 分组名数组，去重后的 group 列表
// DEFAULT_FIELDS: 预设默认列（必须含 type 属性）
```

### 前端核心代码

```javascript
// ===== 渲染字段树 =====
function renderFieldTreeByGroup(group) {
    var q = ($("#fieldSearch").val() || "").toLowerCase();
    var html = "";
    for (var g = 0; g < FIELD_GROUPS.length; g++) {
        var gname = FIELD_GROUPS[g];
        if (!q && gname !== group) continue;
        html += '<div style="font-weight:600;padding:4px 6px;background:#f5f5f5">' + gname + "</div>";
        for (var i = 0; i < FIELD_META.length; i++) {
            var f = FIELD_META[i];
            if (f.group !== gname) continue;
            if (q && f.title.toLowerCase().indexOf(q) === -1) continue;
            var checked = isFieldSelected(f.field) ? " checked" : "";
            html += '<div><label><input type="checkbox" value="' + f.field + '"' + checked
                 + ' onclick="onFieldCheck(this)"> ' + f.title + "</label></div>";
        }
    }
    $("#fieldTree").html(html);
}

// ===== 拖拽排序 (IE11 HTML5 Drag API) =====
var selDragField = "";
function onSelDragStart(e) {
    e = e || window.event;
    selDragField = e.target.getAttribute("data-field") || "";
    e.dataTransfer.effectAllowed = "move";
    e.dataTransfer.setData("text", selDragField);
    e.target.style.opacity = "0.4";
}
function onSelDrop(e) {
    e = e || window.event; e.preventDefault();
    var toField = e.target.getAttribute("data-field") || "";
    var fromField = e.dataTransfer.getData("text") || selDragField;
    // splice 数组重排
    var fromIdx = selectedFields.findIndex(function(f){return f.field===fromField;});
    var toIdx = selectedFields.findIndex(function(f){return f.field===toField;});
    var item = selectedFields.splice(fromIdx, 1)[0];
    selectedFields.splice(toIdx, 0, item);
    buildSelectedList();
}

// ===== 兜底：始终从 FIELD_META 获取正确的 type =====
function buildColumnsWithType() {
    var cols = [];
    for (var i = 0; i < selectedFields.length; i++) {
        var f = selectedFields[i];
        var ftype = f.type;
        for (var k = 0; k < FIELD_META.length; k++) {
            if (FIELD_META[k].field === f.field) { ftype = FIELD_META[k].type; break; }
        }
        cols.push({field: f.field, title: f.title, piece: f.piece, type: ftype || "String"});
    }
    return cols;
}
```

### 关键规范

- `DEFAULT_FIELDS` 必须包含 `type` 属性（Date/Number/CTLoc/Gender/String）
- `buildColumnsWithType()` 始终查 `FIELD_META` 获取 type，不依赖 `f.type||"String"` 兜底
- 拖拽排序直接用 IE11 原生 HTML5 Drag API，无需第三方库
- 分组 tabs + 搜索过滤：有搜索词时跨组显示所有匹配字段

## 2. 动态条件引擎

**场景**：查询界面需要灵活配置过滤条件，字段类型自动决定可用运算符和输入控件。

### 输入

```javascript
// FIELD_META 中的字段格式：field|type  (如 "MR_CYRQ|Date")
// conds JSON：[{field, piece, type, op, value, logic, groupId}]
```

### 前端核心代码

```javascript
// ===== 字段选择 -> 自动切换运算符列表 =====
function onCondFieldChange(el, rec) {
    var val = rec ? rec.value : $(el).combobox("getValue");
    var parts = val.split("|");
    var fieldType = parts[1];

    var ops = [];
    if (fieldType === "Date" || fieldType === "Number") {
        ops = [{v:"eq",l:"等于"},{v:"neq",l:"不等于"},{v:"gt",l:"大于"},
               {v:"gte",l:"大于等于"},{v:"lt",l:"小于"},{v:"lte",l:"小于等于"},
               {v:"between",l:"区间"}];
    } else {
        // String / CTLoc / Gender 使用字符串运算符
        ops = [{v:"contains",l:"包含"},{v:"notcontains",l:"不包含"},
               {v:"startswith",l:"左匹配"}];
    }
    // 渲染运算符 select...
}

// ===== 运算符变更 -> 自动切换输入控件 =====
function onCondOpChange(row) {
    var op = row.find(".cond-op").val();
    var fieldType = /* 从 combobox 取值解析 */;
    var isDate = (fieldType === "Date");

    if (op === "between") {
        // 双输入框（日期用 datebox，数字用 numberbox）
        row.find(".cond-value").html(
            '<input class="cond-val1 ' + (isDate?"hisui-datebox":"") + '"> ~ '
          + '<input class="cond-val2 ' + (isDate?"hisui-datebox":"") + '">');
    } else {
        // 单输入框
        row.find(".cond-value").html(
            '<input class="cond-val1 ' + (isDate?"hisui-datebox":"") + '">');
    }
    if (isDate) {
        row.find(".cond-val1").datebox({editable:false});
        if (op === "between") row.find(".cond-val2").datebox({editable:false});
    }
}

// ===== 收集条件 JSON =====
function collectConditions() {
    var conds = [];
    $("#condContainer .cond-group").each(function(gi) {
        $(this).find(".cond-row").each(function(ri) {
            var fieldSel = $(this).find(".cond-field").combobox("getValue");
            var op = $(this).find(".cond-op").val();
            var logic = $(this).find(".cond-logic").val() || "AND";
            var v1 = $(this).find(".cond-val1").val();
            // ... 收集 field/piece/type/op/value/logic/groupId
            conds.push({field, piece, type, op, value, logic, groupId: gi});
        });
    });
    return conds;
}
```

### 运算符规则

| 字段类型 | 可用运算符 |
|---------|-----------|
| String / CTLoc / Gender | contains, notcontains, startswith |
| Date / Number | eq, neq, gt, gte, lt, lte, between |

### 条件组逻辑

- **组内 AND**：同一 `cond-group` 内所有条件必须同时满足
- **组间 OR**：任意一个 `cond-group` 全通过即算匹配
- 后端分组处理：

```objectscript
// 按 groupId 分组条件，组内 | 分隔
for i = 0:1:(condCount-1) {
    s gid = c.%Get("groupId")
    s groups(gid) = groups(gid) _ sep _ field _ "|" _ op _ "|" _ value _ "|" _ piece _ "|" _ type
}
// 遍历匹配
for gi = 1:1:groupCnt {
    s allPass = 1
    for ci = 1:1:condCnt {
        if '..MatchValue(actualVal, op, value, ftype) s allPass = 0 q
    }
    if allPass s groupMatch = 1 q  // 任一组通过即可
}
```

## 3. MatchValue 通用比较器

**场景**：需要将用户输入的查询条件与 Global 中的实际值进行比较，支持多种运算符和类型自动处理。

### 接口定义

```objectscript
/// MatchValue(actualVal, op, condVal, ftype) -> %Boolean
/// actualVal: 从 Global $p 提取的实际值（或 BuildMultiPiece 拼接值）
/// op: contains|notcontains|startswith|eq|neq|gt|gte|lt|lte|between
/// condVal: 用户输入的查询值（Date 类型已 $zdh 转换）
/// ftype: String|Date|Number|CTLoc|Gender
```

### 完整代码

```objectscript
ClassMethod MatchValue(actualVal As %String, op As %String, condVal As %String, ftype As %String = "String") As %Boolean [ Language = objectscript ]
{
    // CTLoc 类型：ID 转描述后再比较
    if ftype = "CTLoc" s actualVal = $p($g(^CTLOC(+actualVal)), "^", 2)

    if op = "contains"    q ($find(actualVal, condVal) > 0)
    if op = "notcontains" q ($find(actualVal, condVal) = 0)
    if op = "startswith"  q ($e(actualVal, 1, $l(condVal)) = condVal)

    if op = "eq" {
        if ftype = "Date" { s condVal = $zdh(condVal,3) }
        q (actualVal = condVal)
    }
    if op = "neq" {
        if ftype = "Date" { s condVal = $zdh(condVal,3) }
        q (actualVal '= condVal)
    }
    if op = "gt" {
        if ftype = "Date" { s condVal = $zdh(condVal,3) }
        q (actualVal > condVal)
    }
    if op = "lt" {
        if ftype = "Date" { s condVal = $zdh(condVal,3) }
        q (actualVal < condVal)
    }
    if op = "gte" {
        if ftype = "Date" { s condVal = $zdh(condVal,3) }
        q (actualVal '< condVal)
    }
    if op = "lte" {
        if ftype = "Date" { s condVal = $zdh(condVal,3) }
        q (actualVal '> condVal)
    }
    if op = "between" {
        s v1 = $p(condVal, "~", 1)
        s v2 = $p(condVal, "~", 2)
        if ftype = "Date" {
            s v1 = $zdh(v1, 3)
            s v2 = $zdh(v2, 3)
        }
        q (actualVal '< v1)&&(actualVal '> v2)
    }
    q 1
}
```

### 调用示例

```objectscript
// 1. 从 Global 提取值
s actualVal = $p(rowData, "^", piece)

// 2. 聚合字段：先拼接
if piece < 0 s actualVal = ..BuildMultiPiece(rowData, piece)

// 3. 条件比较
if '..MatchValue(actualVal, op, value, ftype) s allPass = 0 q
```

### 扩展新类型

```objectscript
// 示例：新增 Gender 类型预处理（实际已在使用）
if ftype = "Gender" {
    if val = "1" s val = "男"
    if val = "2" s val = "女"
}
// 预处理放在所有运算符判断之前
```

## 4. 模板系统

**场景**：需要将用户的配置（列选择+查询条件）保存为命名模板，支持个人/通用范围、权限控制、SQL 查询。

### 存储方案

```
Global 路径: ^User.DHCMRQueryTemplateD
$lb 格式 (10位):
  1. %%CLASSNAME  = "User.DHCMRQueryTemplate"
  2. Name         = "出院患者查询"
  3. Scope        = "G" | "P"
  4. CreateUserID = "user123"
  5. ColumnsJSON  = '[{"field":"MR_XM","title":"姓名",...}]'
  6. ConditionsJSON = '[{"field":"MR_CYRQ","op":"between",...}]'
  7. IsDefault    = 0 | 1
  8. IsActive     = 0 | 1
  9. CreateDate   = +$h
  10. CreateTime  = +$p($h,",",2)
```

### 持久类（SQL 双通道）

```objectscript
Class User.DHCMRQueryTemplate Extends %Persistent
    [ SqlRowIdName = Tpl_Rowid, SqlTableName = DHCMR_QueryTemplate ]
{
Property Name As %String(MAXLEN = 100, MINLEN = 4) [ Required, SqlColumnNumber = 2, SqlFieldName = Tpl_Name ];
Property Scope As %String(MAXLEN = 1, VALUELIST = ",G,P") [ Required, SqlColumnNumber = 3, SqlFieldName = Tpl_Scope ];
Property CreateUserID As %String [ SqlColumnNumber = 4, SqlFieldName = Tpl_CreateUserID ];
Property ColumnsJSON As %String(MAXLEN = "") [ Required, SqlColumnNumber = 5 ];
Property ConditionsJSON As %String(MAXLEN = "") [ Required, SqlColumnNumber = 6 ];
Property IsDefault As %Boolean [ SqlColumnNumber = 7, SqlFieldName = Tpl_IsDefault ];
Property IsActive As %Boolean [ SqlColumnNumber = 8, SqlFieldName = Tpl_IsActive ];
Property CreateDate As %Date [ SqlColumnNumber = 9 ];
Property CreateTime As %Time [ SqlColumnNumber = 10 ];

Storage Default {
    <Data name="DHCMRQueryTemplateDefaultData">
    <Value name="1"><Value>%%CLASSNAME</Value></Value>
    <Value name="2"><Value>Name</Value></Value>
    ...
    </Data>
    <DataLocation>^User.DHCMRQueryTemplateD</DataLocation>
    <IdLocation>^User.DHCMRQueryTemplateD</IdLocation>
    ...
}
}
```

### 模板加载恢复

```javascript
// 时序级联恢复条件组 (避免 combobox/datebox 异步渲染冲突)
function restoreConditions(conds) {
    // 150ms: 设置字段值 + 触发 onCondFieldChange
    setTimeout(function(row, fieldVal) {
        row.find(".cond-field").combobox("setValue", fieldVal);
        onCondFieldChange(row.find(".cond-field")[0], {value: fieldVal});
    }, 150, lastRow, fv);

    // 350ms: 设置运算符 + trigger change
    setTimeout(function(row, opVal) {
        row.find(".cond-op").val(opVal).trigger("change");
    }, 350, lastRow, opVal);

    // 600ms: 填充值 (datebox 用 setValue)
    setTimeout(function(row, val1, val2) {
        if (isDate) row.find(".cond-val1").datebox("setValue", val1);
        else row.find(".cond-val1").val(val1);
    }, 600, lastRow, v1, v2);
}
```

### SQL 双通道检查清单

| 检查项 | 说明 |
|--------|------|
| 显式 Storage | DataLocation + IdLocation（均 root） + Data 段 | 
| IdLocation | `$i(^Global)` 非 `$i(^Global(0))` |
| Data 段 | 每个 property 精确映射 piece |
| Property | SqlColumnNumber + SqlFieldName |
| Class | SqlRowIdName + SqlTableName |
| $lb 位置 | 位置1=%%CLASSNAME，属性从2开始 |
| 数值写入 | `+$p` 强转，禁止直接字符串写入 |

## 5. 类型转换链

**场景**：Global 中存储的是原始值（ID、$H日期、数字），展示给用户时需要转换为可读格式。

### 转换触发点

在 QueryData 输出循环中，根据字段类型逐一转换：

```objectscript
for ci = 0:1:(colCount-1) {
    s col = colArr.%Get(ci)
    s piece = col.%Get("piece")
    s fld = col.%Get("field")

    // === 取值 ===
    if piece < 0 {
        s val = ..BuildMultiPiece(rowData, piece)    // 聚合字段
    } else {
        s val = $p(rowData, "^", piece)
    }
    if val = "" s val = "-"

    // === 类型转换 ===
    if col.%Get("type") = "Date" {
        if +val > 0 s val = $zd(+val, 3)             // $H -> yyyy-MM-dd
    }
    if col.%Get("type") = "CTLoc" {
        s locDesc = $p($g(^CTLOC(+val)), "^", 2)     // ID -> 科室名
        if locDesc '= "" s val = locDesc
    }
    if col.%Get("type") = "Gender" {
        if val = "1" s val = "男"                      // 1 -> 男
        if val = "2" s val = "女"                      // 2 -> 女
    }
    // + 更多类型扩展...
}
```

### 类型转换速查

| 类型 | 存储值 | 显示值 | 转换函数 |
|------|--------|--------|---------|
| Date | 67761 ($H) | 2026-04-01 | `$zd(+val, 3)` |
| Date(条件) | "2026-04-01" (用户输入) | 67761 ($H) | `$zdh(condVal, 3)` |
| CTLoc | 35 (ID) | 内科 | `$p(^CTLOC(+val),"^",2)` |
| CTLoc(条件) | "内科" (用户输入) | — | MatchValue 内 actualVal 先转描述 |
| Gender | 1 | 男 | 直接映射 |
| Gender | 2 | 女 | 直接映射 |
| MultiPiece | -1 | 所有诊断拼接 | `BuildMultiPiece(data, -1)` |
| MultiPiece | -2 | 所有诊断名称拼接 | `BuildMultiPiece(data, -2)` |
| MultiPiece | -3 | 所有手术拼接 | `BuildMultiPiece(data, -3)` |
| MultiPiece | -4 | 所有手术名称拼接 | `BuildMultiPiece(data, -4)` |

### MultiPiece 实现

```objectscript
ClassMethod BuildMultiPiece(data As %String, piece As %Integer) As %String
{
    s result = ""
    if piece = -1 {
        // 出院主要诊断(52) + 出院其他诊断1-15
        for p = 52,56,60,64,68,72,76,80,84,88,92,96,100,104,108,112 {
            s result = result _ $p(data, "^", p) _ " "
        }
        q $zstrip(result, "<>W")    // 去首尾空格
    }
    // piece = -2/-3/-4 同理...
    q result
}
```

### 条件匹配中的类型处理

```objectscript
// CTLoc 条件：actualVal(ID) 先转描述再与用户输入的描述比较
if ftype = "CTLoc" s actualVal = $p($g(^CTLOC(+actualVal)), "^", 2)

// Date 条件：用户输入(2026-04-01) 先 $zdh 转为 $H 再比较
if ftype = "Date" { s condVal = $zdh(condVal, 3) }
```

### 关键陷阱

1. **`$p` 始终返回字符串** — `$p($h,",",2)` 返回 `"81923"` 非 `81923`，写入 `$lb` 前必须 `+` 强转
2. **`$zd/$zdh` 参数必须是数字** — `$p` 返回值需 `+val` 强转：`$zd(+val, 3)`
3. **`^CTLOC` 不是 `^CT("LOC")`** — HIS 标准 Global 名是 `^CTLOC`，查科室描述用 `$p(^CTLOC(id),"^",2)`
4. **`buildColumnsWithType()` 始终查 `FIELD_META`** — 不依赖 `f.type||"String"` 兜底，防止旧模板/DEFAULT_FIELDS 缺失 type 导致转换被跳过
