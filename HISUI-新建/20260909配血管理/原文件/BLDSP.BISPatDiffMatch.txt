Class BLDSP.BISPatDiffMatch Extends %RegisteredObject
{

/// Creator：      baoguotao
/// CreatDate：    20200819
/// Description:： 血液销毁保存
/// Table：        
/// Input：        Param参数内容XML格式
/// 			   Param.P0 
/// Output：       无
/// Return：       
/// Others: w ##Class(BLDSP.BISPatDiffMatch).SavePatDiffMatch("00011","<Data><P3>1</P3><P12></P12><P4></P4><P0>3</P0><P8>23:00</P8><P10></P10><P5>0</P5><P1>2</P1><P13></P13><P9></P9><P14></P14><P6>0</P6><P2>1</P2><P11></P11><P7>2018-10-11</P7></Data>","113^26^^2^1")	  
ClassMethod SavePatDiffMatch(RegNo As %String, Param As %String(MAXLEN=32767), Sessions As %String, Output RetValue As %String(MAXLEN=32767)) As %String [ SqlProc ]
{
	//s ^tmpsavepatdiff=$Lb(RegNo,Param,Sessions)
	s Sessions=$g(Sessions)
	s WorkGroupDR=$p(Sessions,"^",2)
	s UserDR=$p(Sessions,"^",1) 
	s HospitalDR=$p(Sessions,"^",5) 
	Set $ZTrap = "ErrorHandle"
    s err="",RetValue=""
    Set reader = ##class(%XML.Reader).%New()  
    Do reader.OpenString(Param)
    D reader.Correlate("Data","LIS.WS.Model.InputParameter")  
    While (reader.Next(.obj,.sc)) { 
	    If ($SYSTEM.Status.IsOK(sc)) // check if this succeeded without errors
	    { 
		    s PatDMRecordDR=$tr(obj.P0,$c(0))
		    s DifficultMatchDR= $tr(obj.P1,$c(0))
		    s ReqFormDR=$tr(obj.P2,$c(0))
		    s Remark= $tr(obj.P3,$c(0)) 
	    }
    } 
	D reader.Close()  
	i '$l(DifficultMatchDR) s RetValue="-1^传入患者疑难类型不能为空" q RetValue
	 
		
 	TSTART 
	  i $l(PatDMRecordDR) s objPatDifficultMatchRecord=##class(dbo.BDPatDifficultMatchRecord).%OpenId(PatDMRecordDR)
	  e  s objPatDifficultMatchRecord=##class(dbo.BDPatDifficultMatchRecord).%New()
	  s PreDifficultMatchDR=""
	  i $l(PatDMRecordDR){
         i objPatDifficultMatchRecord.DifficultMatchDR'=DifficultMatchDR{
          s PreDifficultMatchDR=objPatDifficultMatchRecord.DifficultMatchDR
         }
      }  
	  s objPatDifficultMatchRecord.AddDate=$zd($h,8)
	  s objPatDifficultMatchRecord.AddTime=$p($h,",",2)
	  s objPatDifficultMatchRecord.AddUserDR=UserDR
	  s objPatDifficultMatchRecord.DifficultMatchDR=DifficultMatchDR
	  s objPatDifficultMatchRecord.Remark=Remark
	  s objPatDifficultMatchRecord.RegNo=RegNo
	  s objPatDifficultMatchRecord.ReqFormDR=ReqFormDR
	  s sc=objPatDifficultMatchRecord.%Save()
	   If ($SYSTEM.Status.IsOK(sc)) {
	     s RetValue=1
       }
       else {
	     s RetValue=$SYSTEM.Status.GetErrorText(sc) q
       }
       i $l(PreDifficultMatchDR){ 
         d ##class(BLD.Common.BDSYSModifyLog).SaveBDSYSModifyLogMTHD("dbo.BDPatDifficultMatchRecord",PatDMRecordDR,"DifficultMatchDR",PreDifficultMatchDR,DifficultMatchDR,UserDR,WorkGroupDR)
       } 
 
	i RetValue'=1 TROLLBACK  Q RetValue 
    TCOMMIT
	Q RetValue 

ErrorHandle 
    TROLLBACK
	s RetValue="-1^错误"_$tr($ZERROR,"^","--")_".错误代码:"_$ECODE
    Quit RetValue
}

ClassMethod DelPatDiffMatch(RegNo As %String, Param As %String(MAXLEN=32767), Sessions As %String, Output RetValue As %String(MAXLEN=32767)) As %String [ SqlProc ]
{
 
	s Sessions=$g(Sessions)
	s WorkGroupDR=$p(Sessions,"^",2)
	s UserDR=$p(Sessions,"^",1) 
	s HospitalDR=$p(Sessions,"^",5) 
	Set $ZTrap = "ErrorHandle"
    s err="",RetValue=""
    Set reader = ##class(%XML.Reader).%New()  
    Do reader.OpenString(Param)
    D reader.Correlate("Data","LIS.WS.Model.InputParameter") 
    s fPackDRs=""
    While (reader.Next(.obj,.sc)) { 
	    If ($SYSTEM.Status.IsOK(sc)) // check if this succeeded without errors
	    { 
		    s PatDMRecordDR=$tr(obj.P0,$c(0)) 
	    }
    } 
	D reader.Close()  
	i '$l(PatDMRecordDR) s RetValue="-1^传入患者疑难记录不能为空" q RetValue
	 
		
 	TSTART
       s sc=##class(dbo.BDPatDifficultMatchRecord).%DeleteId(PatDMRecordDR) 
	   If ($SYSTEM.Status.IsOK(sc)) {
	     s RetValue=1
       }
       else {
	     s RetValue=$SYSTEM.Status.GetErrorText(sc) q
       }
       s TableName=##class(LIS.Util.Common).IndexData("dbo.BDPatDifficultMatchRecord")
       
       s SYSModifyLogDR="" 
       for{ 
	       s SYSModifyLogDR=$o(^dbo.BDSYSModifyLogI("IndexMaster",TableName,PatDMRecordDR,SYSModifyLogDR)) 
	       q:SYSModifyLogDR=""
	       s sc=##class(dbo.BDSYSModifyLog).%DeleteId(SYSModifyLogDR) 
	       If ($SYSTEM.Status.IsOK(sc)) {
	        s RetValue=1
           }
           else {
	          s RetValue=$SYSTEM.Status.GetErrorText(sc) q
           } 
       
       }   
	 i RetValue'=1 TROLLBACK  Q RetValue 
     TCOMMIT
	 Q RetValue 

ErrorHandle 
    TROLLBACK
	s RetValue="-1^错误"_$tr($ZERROR,"^","--")_".错误代码:"_$ECODE
    Quit RetValue
}

}
