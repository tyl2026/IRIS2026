ClassMethod QryCheckUserExecute(ByRef qHandle As %Binary, Code, P1, P2, P3, P4, P5, P6, P7, P8, P9, P10, P11, P12, P13, Sessions, Output RowCount As %String) As %Status
{
	
	Set repid=$I(^CacheTemp)
 	s ind=1
    S RowCount=0
 	s qHandle=$lb(0,repid,0)
 	Quit:$g(Code)="" $$$OK
	S Rowid=$O(^dbo.SYSUserI("IndexCode",##Class(LIS.Util.Common).IndexData(Code),""))
	i Rowid="" q 1
	
	s UserData=$g(^dbo.SYSUserD(Rowid))
    s CName=$lg(UserData,3) 
    i $l(CName) s CName=CName_"("_Code_")"
	d OutPut
	Quit $$$OK 
	

OutPut
	Set Data=$lb(Code,CName)
        Set ColFields="Code,CName"
 	Set ^CacheTemp(repid,ind)=Data
        Set RowCount=RowCount+1
 	Set ind=ind+1
	q
}

ClassMethod QryCheckUserClose(ByRef qHandle As %Binary) As %Status [ PlaceAfter = QryCheckUserExecute ]
{
	Set repid=$li(qHandle,2)
 	Kill ^CacheTemp(repid)
	Quit $$$OK
}

ClassMethod QryCheckUserFetch(ByRef qHandle As %Binary, ByRef Row As %List, ByRef AtEnd As %Integer = 0) As %Status [ PlaceAfter = QryCheckUserExecute ]
{
	Set AtEnd=$li(qHandle,1)
 	Set repid=$li(qHandle,2)
 	Set ind=$li(qHandle,3)
 	
 	Set ind=$o(^CacheTemp(repid,ind))
 	If ind="" {				
 		Set AtEnd=1
 		Set Row=""
 	}
 	else      {			
 		Set Row=^CacheTemp(repid,ind)
 	}
 	Set qHandle=$lb(AtEnd,repid,ind)
	Quit $$$OK
}

ClassMethod QryIssuePackClose(ByRef qHandle As %Binary) As %Status [ PlaceAfter = QryIssuePackExecute ]
{
	Set repid=$li(qHandle,2)
 	Kill ^CacheTemp(repid)
	Quit $$$OK
}

ClassMethod QryIssuePackFetch(ByRef qHandle As %Binary, ByRef Row As %List, ByRef AtEnd As %Integer = 0) As %Status [ PlaceAfter = QryIssuePackExecute ]
{
	Set AtEnd=$li(qHandle,1)
 	Set repid=$li(qHandle,2)
 	Set ind=$li(qHandle,3) 	
 	Set ind=$o(^CacheTemp(repid,ind))
 	If ind="" {				
 		Set AtEnd=1
 		Set Row=""
 	}
 	else      {			
 		Set Row=^CacheTemp(repid,ind)
 	}
 	Set qHandle=$lb(AtEnd,repid,ind)
	Quit $$$OK
}
