/// w ##class(web.DHCLYLDZN.Method.Tools).Getorginfo()

{

	///批次号由医疗机构代码-填报日期(格式为YYYYMMDD)-当天批次(两位数字)-上传类型(1新增；2重传)

/// w ##class(web.DHCLYLDZN.Method.Tools).GetorgbatchnoByDate()
ClassMethod GetorgbatchnoByDate() As %String
{
	///H43010400157-20250605-0-1
	///批次号由医疗机构代码-填报日期(格式为YYYYMMDD)-当天批次(两位数字)-上传类型(1新增；2重传)
	S dqDate=$zd(+$h,8)
	s ^OrgbatchnoByDate(+$H)="H43110300029-"_dqDate_"-"_$g(^OrgLYLD(+$h,"batchno"))_"-"_$g(^OrgLYLD("XGBZ"))
	q ""
}

/// 重传设置
/// w ##class(web.DHCLYLDZN.Method.Tools).SetReUpINFO()
ClassMethod SetReUpINFO(ReUpflag = "") As %String
{
	
	//初始化调用位置：web.DHCLYLDZN.Method.JCXX
	k ^OrgLYLD
	s SDate=""
	
	//k ^OrgLYLDBC(+$h)
	//补传日期和批次流水号设置，单日期补传,^OrgLYLDBC保存每日补传最大批次流水号
	//补传日期1.1-1.11，2.2-2.23,3.3-3.15
	//补传日期4.3-4.5  4.11  4.27-4.28
	s SDate="2026-03-26"
				
	
	s EDate=SDate      
	s bcbatchno="02"    //补传批次流水号号02开始递增
	i $d(^OrgLYLDBC(+$h,"bcbatchno")) d
	.i +(^OrgLYLDBC(+$h,"bcbatchno"))>8  s ^OrgLYLDBC(+$h,"bcbatchno")=$g(^OrgLYLDBC(+$h,"bcbatchno"))+1
	.e  s ^OrgLYLDBC(+$h,"bcbatchno")="0"_(+$g(^OrgLYLDBC(+$h,"bcbatchno"))+1)
	e  s ^OrgLYLDBC(+$h,"bcbatchno")=bcbatchno
	
	
	i (SDate = "")||(EDate = "") d
	.s ^OrgLYLD("SDate")=$zd(+$h-1,3)
	.s ^OrgLYLD("EDate")=$zd(+$h-1,3)
	.s ^OrgLYLD("SDate2")=$zd(+$h,3)
	.s ^OrgLYLD("EDate2")=$zd(+$h,3)
	.s ^OrgLYLD("XGBZ")=1
	.s ^OrgLYLD(+$h,"batchno")="01"
	e  d
	.s ^OrgLYLD("SDate")=SDate
	.s ^OrgLYLD("EDate")=EDate
	.s ^OrgLYLD("SDate2")=SDate
	.s ^OrgLYLD("EDate2")=EDate
	.s ^OrgLYLD("XGBZ")=1
	.s ^OrgLYLD(+$h,"batchno")=$g(^OrgLYLDBC(+$h,"bcbatchno"))
	.i ReUpflag'="" s ^OrgLYLD("XGBZ")=2  //重传传2
	D ##class(web.DHCLYLDZN.Method.Tools).GetorgbatchnoByDate()   ///产生批次号
	q ""
}

    .s platformtablename=""

    ..s expectedtotalcount=+$g(^TOrgbatchnoByDate(+$h,orgbatchno,platformtablename,"TNum"))

    ..s detailuploadflag=1
    ..s statisticsflag=2
    ..s securitylevel="000"
    ..s valiflag=1
    ..s createtime=$zd(+$h,3)_" "_$zt($p($h,",",2),1)
    ..s updatetime=createtime
	..d OutputRow 
	
	Quit $$$OK
OutputRow                             
	s data=$lb(orgcode,platformtablename,processstarttime,processendtime,expectedtotalcount,actualuploadcount,detailuploadflag,statisticsflag,securitylevel,valiflag,createtime,updatetime,orgbatchno)
    s ^CacheTemp(repid,ind)=data

    ..s batchesstatus=1
	..d OutputRow

	d OutputRow
	s ^TOrgbatchnoByDate(+$H,orgbatchno,"TBL_BATCHES","UPTime")=$zd(+$h,3)_" "_$zt($p($h,",",2),1)
	s:'$d(^TOrgbatchnoByDate(+$H,orgbatchno,"TBL_BATCHES","TNum")) ^TOrgbatchnoByDate(+$H,orgbatchno,"TBL_BATCHES","TNum")=0
	Quit $$$OK
OutputRow                            
	s ^TOrgbatchnoByDate(+$H,orgbatchno,"TBL_BATCHES","TNum")=ind 
	s data=$lb(orgcode,tablename,orgbatchno,reportdate,uploaddate,batchesstatus)
    s ^CacheTemp(repid,ind)=data
	s ind=ind+1

	q
}

/// Description:：     根据就诊号获取有效的InstanceIDList
/// Table：            
/// Input：         Adm:就诊号
/// Output：         获取该患者所有的有效的病例实例ID列表，以“^”作为分割符
/// Return：         
/// Others：        w ##class(MKLCYX.LCYXInterface).GetALLInstanceID("136419")
ClassMethod GetALLInstanceID(Adm As %String) As %String
{
    q:($d(Adm)=0)||(Adm="") ""
    
    s (InstanceIDList,ValidInstanceIDList) = ""
    s DocID = "66"
       s TheECRecord=""
    For {
        
        ;s DocID = $O(^DHCEMRI.ECRecordI("IdxEpisodeIDChartItemID",Adm,DocID,""))
        ;q:(DocID="")
        s TheECRecord = $O(^DHCEMRI.ECRecordI("IdxEpisodeIDChartItemID",Adm,DocID,TheECRecord))
        Q:(TheECRecord="")
        
        s LisNo = ""
        For{
           
            s InstanceID = ""
            s LisNo = $O(^DHCEMRI.InstanceDataI("IdxEcRecordAndListNo",TheECRecord,LisNo))
            q:(LisNo="")
            s InstanceID = $TR(TheECRecord_"||"_LisNo," ","")
            s objInstance = ##class(EMRinstance.InstanceData).%OpenId(InstanceID)
            continue:($IsObject(objInstance)=0)
            s status = objInstance.Status
            continue:(status'="Save")
            // 过滤因同一模板挂到两个界面展示下导致的数据重复
            s tempInstanceID = "^"_InstanceID_"^"
            continue:(ValidInstanceIDList[tempInstanceID)
            s ValidInstanceIDList = ValidInstanceIDList_tempInstanceID
            
            s:(InstanceIDList'="") InstanceIDList = InstanceIDList_"^"
            s InstanceIDList = InstanceIDList_InstanceID
        }
    }
    q InstanceIDList
}
