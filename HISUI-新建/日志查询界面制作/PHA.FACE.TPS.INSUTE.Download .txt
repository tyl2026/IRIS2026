                            Set data = recObj.data
                            Set code = recObj.code
                            Set msg = recObj.msg
                            If (+code<0) {
                                If (msg = "") {
                                    Set msg = recObj.info    
                                }
                                Set exit=1,errmsg="入库失败:" _ msg
                                Trollback
                                Quit
                            }
                            /// 入库成功, 处理追溯码
                            Set ingr = data
                            If (ingrStr = "") {
                                Set ingrStr = ingr
                            }Else {
                                Set ingrStr = ingrStr + "^" + ingr
                            }
                        }Else {
                            Set ingrStr = ingr
                        }
                        Set itm = 0
                        For {
                            Set itm = $ORDER(^DHCINGR(ingr,"GRI",itm))    Quit:(itm = "")||(exit=1)
                            Set ingrItm = ingr _"||"_ itm
                            Set intr = +$ORDER(^DHCINTR(0,"TypePointer","G",ingrItm,0))
                            Set inclb = $PIECE(^DHCINTR(intr),"^",7)
                            Set rp = $PIECE(^DHCINTR(intr),"^",16)
                            Set incib = $PIECE(^INCI(+inclb,"IL",$PIECE(inclb,"||",2),"LB",$PIECE(inclb,"||",3)),"^",1)
                            Set intrBatNo = $PIECE(^INCI(+incib,"IB",$PIECE(incib,"||",2)),"^",1)
                            Set intrExp = $PIECE(^INCI(+incib,"IB",$PIECE(incib,"||",2)),"^",2)
                            Set inciManf = $PIECE(..GetManfNew(inci),"^",1)
                            Set scgCat = $p(##Class(web.DHCST.Common.DrugInfoCommon).GetArcItemCat(inci),"^",2)
                            // 取码存
                            If $DATA(TraceCodeARRAY(pid,mainNum,"BTC",+inclb,inciManf,intrBatNo,intrExp,rp)) {
                                    Set btcStr = TraceCodeARRAY(pid,mainNum,"BTC",+inclb,inciManf,
                                    intrBatNo,intrExp,rp)
                                    Set btcLen = $LENGTH(btcStr,"^")
                                    For b = 1:1:btcLen {
                                        Set tempBtc = $PIECE(btcStr,"^",b)
                                        //b:tempBtc="87558700000881278564"  ; 12345
                                        If ($$$comMemberDefined("PHA.DTC.Scan.Save", "m", "SaveTCode")) {
                                            Set saveBtcRet = ##class(PHA.DTC.Scan.Save).SaveTCode(intr, tempBtc, "G", "BTC", User, "")
                                        }Else {
                                            Set saveBtcRet = ##class(PHA.IN.TraceCode.V2.Save).SaveTCode(intr, tempBtc, "G", "BTC", User, "")
                                        }
                                        Set saveCode = $PIECE(saveBtcRet,"^",1)
                                        q:(saveCode < 0)
                                    }
                                    If (saveCode < 0) {
                                        Set exit=1,errmsg="-301^保存追溯码大码失败，原因为:" _ $PIECE(saveBtcRet,"^",2) _",检查配送流水是否已入过库,清手动删除,文件路径为:"_Path
                                        Trollback
                                        Quit
                                    }
                            }
                            If $DATA(TraceCodeARRAY(pid,mainNum,"MTC",+inclb,inciManf,intrBatNo,intrExp,rp)) {
                                    Set mtcStr = TraceCodeARRAY(pid,mainNum,"MTC",+inclb,inciManf,intrBatNo,intrExp,rp)
                                    Set mtcLen = $LENGTH(mtcStr,"^")
                                    For m = 1:1:mtcLen {
                                        Set tempMtc = $PIECE(mtcStr,"^",m)
                                        //b:tempMtc="87558700000881278564"  ; 1234
                                        If ($$$comMemberDefined("PHA.DTC.Scan.Save", "m", "SaveTCode")) {
                                            Set saveBtcRet = ##class(PHA.DTC.Scan.Save).SaveTCode(intr, tempMtc, "G", "MTC", User, "")
                                        }Else {
                                            Set saveBtcRet = ##class(PHA.IN.TraceCode.V2.Save).SaveTCode(intr, tempMtc, "G", "MTC", User, "")
                                        }
                                        Set saveCode = $PIECE(saveBtcRet,"^",1)
                                        q:(saveCode < 0)
                                    }
                                    If (saveCode < 0) {
                                        Set exit=1,errmsg="-301^保存追溯码中码失败，原因为:" _ $PIECE(saveBtcRet,"^",2) _",检查配送流水是否已入过库,清手动删除,文件路径为:"_Path
                                        Trollback
                                        Quit
                                    }
                            }
                            If $DATA(TraceCodeARRAY(pid,mainNum,"STC",+inclb,inciManf,intrBatNo,intrExp,rp)) {
                                    Set stcStr = TraceCodeARRAY(pid,mainNum,"STC",+inclb,inciManf,
                                    intrBatNo,intrExp,rp)
                                    Set stcLen = $LENGTH(stcStr,"^")
                                    For s = 1:1:stcLen {
                                        Set tempStc = $PIECE(stcStr,"^",s)
                                        Set tempStcBarId = +$o(^BS.PHA.IN.BarCodeI("BarCode", tempStc, ""),-1)
                                        // b:tempStc="83023930157407567512"  
                                        //b ; 1
                                        continue:tempStcBarId'=0
                                        /*                       ; 112
                                        if (tempStcBarId '= 0) {
                                            s comFlag = $lg(^BS.PHA.IN.BarCodeD(tempStcBarId),16)
                                            continue:comFlag=1
                                        }
                                        */
                                        If ($$$comMemberDefined("PHA.DTC.Scan.Save", "m", "SaveTCode")) {
                                            Set saveBtcRet = ##class(PHA.DTC.Scan.Save).SaveTCode(intr, tempStc, "G", "STC", User, "")
                                        }Else {
                                            Set saveBtcRet = ##class(PHA.IN.TraceCode.V2.Save).SaveTCode(intr, tempStc, "G", "STC", User, "")
                                        }
                                        Set saveCode = $PIECE(saveBtcRet,"^",1) 
                                        q:(saveCode < 0)
                                        if (HandleFalg = "Y") { // 为了补码特殊处理标志.  录入成功,直接拆零
                                            // 生成拆零流水
                                            continue:tempStcBarId'=0       // 已经录入了的一定不要再处理
                                            //b ; 122
                                            s serialNo = ##class(web.DHCST.Common.AppCommon).GetAppNo("PHATRACECODE", "", LocId)
                                            if (serialNo = "") {
                                                s serialNo = 3563_"-"_LocId_"-"_+$h_$p($h,",",2)
                                            }else {
                                                s serialNo = 3563_"-"_serialNo
                                            }
                                            s serialBarId = +$o(^BS.PHA.IN.BarCodeI("BarCode", tempStc, ""),-1)
                                            continue:serialBarId=0
                                            if (scgCat [ "注射")||(scgCat [ "输液")||(scgCat [ "针剂") {
                                            	s ret = ##class(PHA.DTC.Scan.Save).SplitByBarId(serialBarId,235,serialNo)
                                            }else {
	                                        	s ret = ##class(PHA.DTC.Scan.Save).SplitByBarId(serialBarId,121,serialNo)
	                                        }
                                        }
                                    } 
                                    If (saveCode < 0) {
                                        Set exit=1,errmsg="-301^保存追溯码小码失败，原因为:" _ $PIECE(saveBtcRet,"^",2) _",检查配送流水是否已入过库,清手动删除,文件路径为:"_Path
                                        Trollback
                                        Quit
                                    }
                            }
                        }
                    }
                }
            }
        }
    }
    If exit=0 {
        Tcommit
        Quit ingrStr
    }Else {
        If $TLEVEL>0 Trollback
        Quit errmsg
    }
HandleHHimportError
    If $TLEVEL>0 Trollback
    Set $ZTRAP="" 
    Quit $ZERROR
}

        Set autoRecepObj = {}
        Set autoRecepObj.initId = initId
        Set autoRecepObj.nextStatusCode = "INAUDIT"
        Set autoRecepObj.userId = transAckUser   // trflocUser

            Set dhcPhaLoc = $ORDER(^DHCPL(0,"Loc",initToLocId,""))
            Set:dhcPhaLoc'="" groupId = 38
            If (groupId="") {
                Set dhcPhLoc = $ORDER(^DHCPHLOCi("LOC",initToLocId,""))
                Set:dhcPhLoc'="" groupId = 34

        }

        Set autoRecepObj.groupId = groupId
        Set retsult = ##class(PHA.IN.TRANS.Api).AuditForIn(autoRecepObj)
        ;b ; 入库明细全部失败  5
        Set resultCode = retsult.code
        If (resultCode'=0) {
            Set ret.code = resultCode
            Set ret.info = "-4^自动接收失败"_$GET(retsult.msg)
            Quit ret
        }
    }
    If (ret.code)>=0 {
        /// 更新订单表
        &sql(UPDATE SQLUser.DHC_HHImport
            SET DHI_flag = 'Y',
            DHI_Vend_DR = :VenId,
            DHI_RecLoc_DR = :trfloc
            WHERE DHI_RowID = :DHIRowId)
        Quit:SQLCODE<0 SQLCODE_"^"_%msg
        Quit ingdret
    }
    If (priceDiffStr'="") {
        Set ret.info = priceDiffStr_"药品价格与当前系统价格不一致，请检查！"
    }
    Quit ret
}

/// Description:生成追溯码关联关系表数据
/// Creator:    hulihua
/// CreateDate: 2026-05-13
/// Table:      BS_PHA_IN.BarCodeRela
/// Input:      tracecode-追溯码，inputLB-信息链表, parentId-父表ID
/// Output:
/// Return:     失败:-1^失败原因，成功:码关联关系表ID
/// Others:     信息链表(码等级、批准文号、批号、效期、生产日期)
/// Debug:      w ##class(PHA.FACE.TPS.INSUTE.Download).InsertBarCodeRela()
ClassMethod InsertBarCodeRela(tracecode, inputLB, parentId = "") As %String
{
    s bcrId = +$o(^BS.PHA.IN.BarCodeRelaI("codeIndex", tracecode, 0))
    q:(bcrId > 0) bcrId
    s expDate = $lg(inputLB, 4)
    if (expDate '= "") {
        if ($F(expDate, "-")) {
            s expDate = $zdh(expDate, 3)
        } else {
            s expDate = $zdh(expDate, 8)
        }
    }
    s producedate = $lg(inputLB, 5)
    if (producedate '= "") {
        if ($F(producedate, "-")) {
            s producedate = $zdh(producedate, 3)
        } else {
            s producedate = $zdh(producedate, 8)
        }
    }
    s barCodeRelaObj = ##class(BS.PHA.IN.BarCodeRela).%New()
    s barCodeRelaObj.barcode = tracecode
    s barCodeRelaObj.codeLevel = $lg(inputLB, 1)
    s barCodeRelaObj.approvNo = $lg(inputLB, 2)
    s barCodeRelaObj.batNo = $lg(inputLB, 3)
    s barCodeRelaObj.expDate = expDate
    s barCodeRelaObj.produceDate = producedate
    d barCodeRelaObj.parentDrSetObjectId(parentId)
    s sc = barCodeRelaObj.%Save()
    if ($$$ISERR(sc)) {
        s errinfo = $System.Status.GetErrorText(sc)
        q "-1^"_errinfo
    }
    q barCodeRelaObj.%Id()
}

/// 保存3563Excel数据中的码关系
/// Debug:      w ##class(PHA.FACE.TPS.INSUTE.Download).SaveCodeRelaBy3563Excel(RetExcelObj)
ClassMethod SaveCodeRelaBy3563Excel(RetExcelObj, ingr) As %Status
{
	Quit:(ingr = "") ..Error(-1, "下载入库单ID不能为空！")
	s dataSize = RetExcelObj.%Size()
    q:dataSize=0 $$$ERROR($$$GeneralError, "无数据")
    s pid = ..NewPid()
    k ^||CacheTemp("PHA","PHA.FACE.TPS.INSUTE.Download",
            "SaveCodeRelaBy3563Excel",pid)
    /// 整理明细数据
    f i=0:1:(dataSize-1) {
        s row = RetExcelObj.%Get(i)
        s insuCode = row."医疗目录编码"
        s tarId = +##class(web.DHCINSUPort).GetTarItmByInsuTarCode2(insuCode,"00A","",2)
        i tarId=0 {
	        ///医保码不对也先把追溯码下下来
	        continue:+ingr=0
	        s tarId=+##class(PHA.FACE.TPS.INSUTE.Download2).getTarByINGR(ingr)
	        s ingrData = $g(^DHCINGR(ingr))
            s approvalNo = $p(ingrData,"^",46)
            s ingrNo = $p(ingrData,"^",1)
            s ^YZSY("SaveCodeRelaBy3563Excel",ingr)=tarId_"^"_ingrNo_"^"_approvalNo_"^"_insuCode_"^"_ $ZD(+$H,3)  //无法通过医保码匹配的，通过子表药品(只有一个药品)匹配的数据存日志
	        }
        
        continue:(tarId = 0)
        s dhcTarId = +$ORDER(^DHCINCTARi("TAR",tarId,""))
        continue:(dhcTarId = 0)
        s inci = +$PIECE(^DHCINCTAR(dhcTarId),"^",1)
        continue:(inci = 0)
        s approveNo = row."批准文号"
        s batNo = row."药品/耗材生产批次号"
        s expDate = row."有效期止"
        s produceDate = row."生产日期"
        s BTCCode = row."大包装溯源码"
        s MTCCode = row."中包装溯源码"
        s STCCode = row."小包装溯源码"
        s codePackLevel = ""   // 码等级(1、2、3, 代表小码、中码、大码)
        s traceCodeList = $lb(codePackLevel, approveNo, batNo, expDate, produceDate)
        s parentBcrId = ""
        //保存大码关系
        s traceCode = BTCCode
        if (traceCode'=STCCode) {
	        d SaveRela("BTC")
        }
        //保存中码关系
        s traceCode = MTCCode
        if (traceCode'=STCCode) {
	        d SaveRela("MTC")
        }
        //保存小码关系
        s traceCode = STCCode
        if (traceCode '= ""){
	        d SaveRela("STC")
        }
    }
    q $$$OK
    
SaveRela(codeType)
	q:($l(traceCode)< 20) 
	s codePackLevel = $case(codeType, "BTC":3, "MTC":2, "STC":1, :"")
	q:(codePackLevel = "")
	s $list(traceCodeList, 1) = codePackLevel
    s parentBcrId = ..InsertBarCodeRela(traceCode, traceCodeList, parentBcrId)
    //保存标识
	s isDrugTCode = ##Class(PHA.DTC.COM.Method).IsDrugTCode(inci, codeType, traceCode)
	if ('isDrugTCode){
		s idCode = ##class(PHA.DTC.COM.Method).GetIDCodeByTCode(traceCode)
		s sCDIDr = $o(^DHCSTSCDI(0, "TYPECODE", "CODETYPE", $$ALPHAUP^SSUTIL4(codeType), ""))
		
		Q:($d(^CT.PHA.IN.IncIDCodeI("TypeActIncCode",sCDIDr,1,inci,idCode)))
		s dataStr = idCode_"^"_sCDIDr_"^"_"Y"
		//s ^PHADEBEBUG("20260814",$zts,inci,dataStr) = $zdt($h, 3)
		d ##class(PHA.DTC.IDCode.Save).Save(inci, "", dataStr,2) 
	}
}
