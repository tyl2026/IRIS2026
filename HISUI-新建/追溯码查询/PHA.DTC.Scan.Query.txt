Class PHA.DTC.Scan.Query Extends %RegisteredObject
{

/// Description:获取业务所对应的台账信息
/// Creator:	suiguyang
/// CreateDate:	20240608
/// Table:      DHC_INTRANS
/// Input:      qBsIdStr-业务表ID字符串, qBsType-业务类型(对应台账类型) 盘点类型取"AT"，qBsShowFlag-标识码为空的是否显示，qCodeType-码类型，qOeoreStr-执行记录串(以^分割)
/// Output:     
/// Return： 
/// Others:     
/// Debug:      d ##class(%ResultSet).RunQuery("PHA.DTC.Scan.Query","GetBsInfo","1853879","F","N","STC")
Query GetBsInfo(qBsIdStr As %String, qBsType As %String, qBsShowFlag As %String, qCodeType As %String, qOeoreStr As %String) As websys.Query(ROWSPEC = "intrStr, inciCode, inciDesc, spec, tQty, pUom, alrPTCNum, alrBTCNum, traceCodeNum, group, needPTCNum, needBTCNum, needTCNum, drugCompleteFlag, traceCodePrefix, inci, tQtyUom, manfName") [ SqlProc ]
{
}

ClassMethod GetBsInfoExecute(ByRef qHandle As %Binary, qBsIdStr As %String, qBsType As %String, qBsShowFlag As %String, qCodeType As %String, qOeoreStr As %String) As %Status
{
	s repid=$I(^CacheTemp)
	s qHandle=$lb(0,repid,0)
	s ind=1
	q:(qBsIdStr = "")&&(qOeoreStr = "") $$$OK
	q:(qBsType = "") $$$OK
	s pid = ..NewPid()
	k GetBsInfo("PHA", $CLASSNAME(), pid) 
	//按执行记录查询
	i (qBsIdStr = "")&&(qOeoreStr '= "") {
		d ##class(PHA.DTC.Nurse.Query).GetDrugInfo(qOeoreStr, qBsShowFlag, qCodeType, pid, .GetBsInfo, qBsType)
	}
	s bsIdStrLen = $l(qBsIdStr, "^")
	for bsIdi = 1 : 1 : bsIdStrLen{		
		s bsId = $p(qBsIdStr, "^", bsIdi)
		//获取对应业务的台账pointer集合
		s pointerArr = ##class(PHA.COM.Array).%New()
		s pointerNum = 
			$CASE(qBsType,
	              "F"  : ..GetFPointerArr(bsId, .pointerArr),   // 门诊发药
	              "H"  : ..GetHPointerArr(bsId, .pointerArr),   // 门诊退药
	              "P"  : ..GetPPointerArr(bsId, .pointerArr),   // 住院发药
	              "Y"  : ..GetYPointerArr(bsId, .pointerArr),   // 住院退药
	              "G"  : ..GetGPointerArr(bsId, .pointerArr),   // 入库
	              "R"  : ..GetRPointerArr(bsId, .pointerArr),   // 退货
	              "A"  : ..GetAPointerArr(bsId, .pointerArr),   // 库存调整
	              "AT" : ..GetATPointerArr(bsId, .pointerArr),  // 盘点       
	              :"0"
	        )
	    continue:(pointerNum = "0")
	    
	    //获取分组信息
	    s group = 
	    	$CASE(qBsType,
	              "F"  : ..GetFGroupInfo(bsId),   // 门诊发药
	              "H"  : ..GetHGroupInfo(bsId),   // 门诊退药
	              "P"  : ..GetPGroupInfo(bsId),   // 住院发药
	              "Y"  : ..GetYGroupInfo(bsId),   // 住院退药
	              "G"  : ..GetGGroupInfo(bsId),   // 入库
	              "R"  : ..GetRGroupInfo(bsId),   // 退货
	              "A"  : ..GetAGroupInfo(bsId),   // 库存调整
	              "AT" : ..GetATGroupInfo(bsId),  // 盘点
	              :""
	        )
	  
	    //利用pointer获取到对应的数据返回前台展示
	    s pointerArrLen=pointerArr.%Size()
		for i = 0 : 1 : (pointerArrLen - 1) {
			s pointer = pointerArr.%Get(i)
			if (qBsType '= "AT") {
				s intr = $o(^DHCINTR(0,"TypePointer", qBsType, pointer, "0"))
				s intrData = ^DHCINTR(intr)
				s inclb = $p(intrData, "^", 7)
				s inci = +inclb
                s traceCodePrefix = ##class(PHA.DTC.COM.Method).GetInciIDCode(inci, qCodeType)
				continue:(qBsShowFlag = "Y")&&(traceCodePrefix = "")
			    s Incib = $P(^INCI(inci,"IL", $p(inclb,"||",2), "LB", $p(inclb,"||",3)), "^", 1)
				s VendManf = ##class(web.DHCST.Common.DrugStkCommon).VendManfByIncIb(Incib) //查找批次供应商及厂家
				s vendor = $p(VendManf, "^", 2)
				s manf = $p(VendManf, "^", 4)
				s tQty = $p(intrData, "^", 6)
				s:tQty<0 tQty = tQty * (-1)
				s intrUomId = $p(intrData, "^", 10)	//台账单位
				s bUomId = $p(^INCI(inci,1), "^", 10)
				s intrFac = ##class(PHA.DTC.COM.Method).UomFac(intrUomId,bUomId)
				s tQty = tQty * intrFac
				s traceCodeBQtyNum = ##class(PHA.DTC.COM.Method).GetRecordBQty(intr,qBsType)  //已绑定的基本单位数量
				d GroupData
			}else{
			
				s insti = $p(pointer, "||", 2)
				s inst = $p(pointer, "||", 1)
				s instiData = $g(^DHCINST(inst,"STI",insti))
				s inclb = $p(instiData,"^",1)
				s inci = +inclb
                s traceCodePrefix = ##class(PHA.DTC.COM.Method).GetInciIDCode(inci, qCodeType)
				continue:(qBsShowFlag = "Y")&&(traceCodePrefix = "")
				s incib = $p(^INCI($p(inclb,"||",1),"IL",$p(inclb,"||",2),"LB",$p(inclb,"||",3)),"^",1)
				s VendManf = ##class(web.DHCST.Common.DrugStkCommon).VendManfByIncIb(incib) //查找批次供应商及厂家
				s vendor = $p(VendManf, "^", 2)
				s manf = $p(VendManf, "^", 4)
				s tQty = $p(instiData,"^",2)
				s:tQty<0 tQty = tQty * (-1)
				s intrUomId = $p(instiData, "^", 17)	// 盘点单位
				s bUomId = $p(^INCI(inci,1), "^", 10)
				s intrFac = ##class(PHA.DTC.COM.Method).UomFac(intrUomId,bUomId)
				s tQty = tQty * intrFac
				s intr = bsId_"||"_insti
				s traceCodeBQtyNum = ##class(PHA.DTC.COM.Method).GetRecordBQty(intr,qBsType)  //已绑定的基本单位数量
				d GroupData
			}
			
			
		}	
	}
	s outIndex=""
	for{
		s outIndex=$o(GetBsInfo("PHA", $CLASSNAME(), pid, outIndex)) q:(outIndex = "")
		s outData= GetBsInfo("PHA", $CLASSNAME(), pid, outIndex)
		s inci = $p(outData, "^", 1)
        s bQty = $p(outData, "^", 2)
		s group = $p(outData, "^", 4)
		s inciCode = $p(^INCI(inci,1), "^", 1)
	    s inciDesc = $p(^INCI(inci,1), "^", 2)
	    s spec = ##class(web.DHCST.Common.DrugInfoCommon).GetSpec("", inci)
	    s bUomId = $p(^INCI(inci,1), "^", 10)
	    s pUomId = $p(^INCI(inci,1), "^", 17)
	    if (pUomId = "") {
			s pUomId = $p(^INCI(inci,3), "^", 6)
	    }
		s pUom = $p(^CT("UOM",pUomId),"^",2)
		s inciFac = ##class(PHA.DTC.COM.Method).UomFac(pUomId, bUomId) 
        s retFac = ##class(PHA.DTC.COM.Method).GetPackFac(inci, qCodeType)
        s tQty = $fn(bQty / inciFac, "", 2)
        s tQtyUom = ##class(PHA.DTC.COM.Method).ToQtyWithUom(inci, bQty)
        s needQty = bQty / retFac		// 需扫数量
        s needPTCNum = +##class(PHA.DTC.COM.Util).MathFloor(needQty)
        s needBTCNum = +$s(+$p(needQty, ".", 2) > 0 : 1, 1 : 0)
        
        s needTCNum = needPTCNum + needBTCNum
        s alrBQty = $p(outData, "^", 3)
        s alrQty = alrBQty / retFac
        s alrPTCNum = +##class(PHA.DTC.COM.Util).MathFloor(alrQty)   	// 已扫数量
        s alrBTCNum = +$s(+$p(alrQty, ".", 2) > 0 : 1, 1 : 0)
        s traceCodeNum = alrPTCNum + alrBTCNum
		s traceCodePrefix=$p(outData, "^", 5)
		s intrStr = GetBsInfo("PHA", $CLASSNAME(), pid, outIndex, "intrStr")
		
		s drugCompleteFlag = ##class(PHA.DTC.COM.Method).IsScanComp(intrStr, qBsType)
		s noTraceCodeFlag = ##class(PHA.DTC.COM.Method).GetIncitmPropValue(inci, "NoTraceCode")
		if (noTraceCodeFlag = "Y") {
			s drugCompleteFlag = "C"
		}
		s manfName = $p(##class(web.DHCST.Common.DrugInfoCommon).GetManf(inci), "^", 3)
		s data = $lb(intrStr, inciCode, inciDesc, spec, tQty, pUom, alrPTCNum, alrBTCNum, traceCodeNum, group ,needPTCNum, needBTCNum, needTCNum, drugCompleteFlag, traceCodePrefix, inci, tQtyUom, manfName)
		s ^CacheTemp(repid,ind) = data
		s ind = ind+1
	}
	k GetBsInfo("PHA", $CLASSNAME(), pid)
	s qHandle=$lb(0,repid,0)
	q $$$OK
GroupData
	s index=group_"^"_inci
	if ('$d(GetBsInfo("PHA", $CLASSNAME(), pid, index))){
		s GetBsInfo("PHA", $CLASSNAME(), pid, index)=inci_"^"_tQty_"^"_traceCodeBQtyNum_"^"_group_"^"_traceCodePrefix
	}else{
		s $p(GetBsInfo("PHA", $CLASSNAME(),pid,index), "^", 2)=tQty + $p(GetBsInfo("PHA", $CLASSNAME(),pid,index), "^", 2)
		s $p(GetBsInfo("PHA", $CLASSNAME(),pid,index), "^", 3)=traceCodeBQtyNum + $p(GetBsInfo("PHA", $CLASSNAME(),pid,index), "^", 3)
		
	}
	if ('$d(GetBsInfo("PHA", $CLASSNAME(), pid, index, "intrStr"))){
		s GetBsInfo("PHA", $CLASSNAME(), pid, index, "intrStr") = intr
	}else{
		s GetBsInfo("PHA", $CLASSNAME(), pid, index, "intrStr") = GetBsInfo("PHA", $CLASSNAME(), pid, index, "intrStr")_"@"_intr
	}
}

/// Description:获取台账对应的追溯码信息
/// Creator:	suiguyang
/// CreateDate:	20240608
/// Table:      DHC_IntrBatPack
/// Input:		qPointer-台账ID
/// Output:		
/// Return：	
/// Others: 	
/// Debug:		d ##class(%ResultSet).RunQuery("PHA.DTC.Scan.Query","GetTraceCodeInfo","1208")
Query GetTraceCodeInfo(qPointerStr As %String, qCodeType As %String) As websys.Query(ROWSPEC = "tpdId,traceCode,codeTypeDesc,scanUserName,trdnFlag,serialNo,matchFlag") [ SqlProc ]
{
}

ClassMethod GetTraceCodeInfoExecute(ByRef qHandle As %Binary, qPointerStr As %String, qCodeType As %String = "") As %Status
{
    s repid=$I(^CacheTemp)
    s qHandle=$lb(0,repid,0)
    s ind=1 
    q:(qPointerStr = "") $$$OK
    s qPointerLen = $l(qPointerStr,"@")
    for qPointeri = 1:1:qPointerLen{
        s qPointer = $p(qPointerStr, "@", qPointeri)
        s tpdId=""
        for {
            s tpdId = $o(^INCTBP("Pointer",qPointer,tpdId),-1) q:(tpdId = "") 
            s tpdData = ^INCTBP(tpdId)
            s traceCode = $p(tpdData,"^",9)
            s scanUserId = $p(tpdData,"^",10)
            s scanUserName = ""
            if (scanUserId '= "") {
                s scanUserName = $p($g(^SSU("SSUSR", +scanUserId)), "^", 2)
            }
            s sCDIDr = $p(tpdData,"^",11)
            s codeTypeCode = $p($g(^DHCSTSCDI(+sCDIDr)), "^", 1)
            continue:(qCodeType '= "")&&(qCodeType '= codeTypeCode)
            s codeTypeDesc = $p($g(^DHCSTSCDI(+sCDIDr)), "^", 2)
            s trdnFlag = $p(tpdData,"^",12)
            s trdnFlag = $s(trdnFlag = "1" : "Y", 1 : "N")
            s barCodeId = $o(^BS.PHA.IN.BarCodeI("BarCode", traceCode, ""))
            s serialNo = $lg($g(^BS.PHA.IN.BarCodeD(+barCodeId)), 18)
            s intrType = $p($G(^DHCINTR(qPointer)),"^",1)
            s matchFlag = ##class(PHA.DTC.COM.Method).IsScanComp(qPointer, intrType)
            s data = $lb(tpdId, traceCode, codeTypeDesc, scanUserName, trdnFlag, serialNo,matchFlag)
            s ^CacheTemp(repid,ind) = data
            s ind = ind+1
        }
    }
    
    s qHandle=$lb(0,repid,0)
    q $$$OK
}

		}

		}

	s patNo = $p(^PAPER(papmi, "PAT",1), "^", 2)                                                         //登记号
    s patName = $p(^PAPER(papmi, "ALL"), "^", 1)                                                         //姓名
	q patNo_"/"_patName
}

/// Description:通过业务指针获取已记录的追溯码个数

/// Others:
/// Debug:		w ##class(PHA.DTC.Scan.Query).GetTraceCodeNum("1208") 	
ClassMethod GetTraceCodeNum(Pointer, qBsType) As %Integer
{
	q:(Pointer = "") ""
	s traceCodeNum=0	
	
	s str = ""
	s traceArr = ##class(PHA.COM.Array).%New()
	s tbpId = ""
	for {
		s tbpId = $o(^INCTBPi("TypePointer",qBsType,Pointer,tbpId))
		q:(tbpId = "")
		continue:('$d(^INCTBP(tbpId)))
		s codeObj = ##class(PHA.COM.Object).%New()
		s barCode = $p(^INCTBP(tbpId), "^", 9)
		s traceCodeNum = traceCodeNum + 1
	}
	q traceCodeNum
}

/// Description:检测某个业务数据是否存在已维护追溯码前缀的库存项
/// Creator:	suiguoyang
/// CreateDate:	20240612
/// Table:      
/// Input:		qBsIdStr-业务表ID字符串，qBsType-业务类型(对应台账类型)，qBsCodeType-码类型
/// Output:		
/// Return：	
/// Others: 	
/// Debug:		w ##class(PHA.DTC.Scan.Query).ChkBsTypePrefix("139^140^141^142","F")

/// Description:检查门诊发药是否包含追溯码前缀
/// Debug:	    w ##class(PHA.DTC.Scan.Query).ChkOPDisp("")
ClassMethod ChkOPDisp(qBsIdStr, qBsCodeType)
{
	s traceCodeFlag = 0
	s bsIdStrLen = $l(qBsIdStr, "^")
	for bsIdi = 1 : 1 : bsIdStrLen{
		q:(traceCodeFlag = 1)
		s bsId = $p(qBsIdStr, "^", bsIdi)
		s phdId = bsId
		s phdCh = ""
		for {
			s phdCh = $o(^DHCPHDI(phdId,"PHDI",phdCh))
			q:(phdCh = "")||(traceCodeFlag = 1)
			continue:'$d(^DHCPHDI(phdId,"PHDI",phdCh))
			s phdSub = ""
			for {
				s phdSub = $o(^DHCPHDI(phdId,"PHDI",phdCh,"INCLB",phdSub))
				q:(phdSub = "")||(traceCodeFlag = 1)
				continue:'$d(^DHCPHDI(phdId,"PHDI",phdCh,"INCLB",phdSub))
				s phdData = ^DHCPHDI(phdId,"PHDI",phdCh,"INCLB",phdSub)
				s inciId = +$p(phdData,"^",3)
                s prefixBar = ##class(PHA.DTC.COM.Method).GetInciIDCode(inciId, qBsCodeType)
				if (prefixBar '= "") {
					s traceCodeFlag = 1
				}
				
			}
		}
	}
	q traceCodeFlag
}

/// Description:获取科室追溯码类型
/// Creator:	hulihua
/// CreateDate:	2024-08-12
/// Table:      
/// Input:		CodeTypeStr-参数设置配置的码类型串
/// Output:		
/// Return：	追溯码类型数据集
/// Others: 	
/// Debug:		w ##class(PHA.DTC.Scan.Query).GetCodeType("")
ClassMethod GetCodeType(CodeTypeStr) As %String
{
	s firCodeType = $p(CodeTypeStr, ",", 1)
	k GetCodeTypeArr
	s scdRowId = ""
	for {
		s scdRowId = $o(^DHCSTSCDI(0, "TYPE", $$ALPHAUP^SSUTIL4("CodeType"), scdRowId))
		q:(scdRowId = "")
		s code = $p(^DHCSTSCDI(scdRowId), "^", 1)
		continue:(CodeTypeStr '= "")&&($lf($lfs(CodeTypeStr), code) = 0)
		s desc = $p(^DHCSTSCDI(scdRowId), "^", 2)
		if (firCodeType '= "")&&(firCodeType = code) {
			s defFlag = 0
		} else {
			s defFlag = 1
		}
		s GetCodeTypeArr(defFlag, scdRowId) = $lb(code, desc)
	}
	s retArr = ##class(PHA.COM.Array).%New()
	s defFlag = ""
	for {
		s defFlag = $o(GetCodeTypeArr(defFlag))
		q:(defFlag = "")
		s scdRowId = ""
		for {
			s scdRowId = $o(GetCodeTypeArr(defFlag, scdRowId))
			q:(scdRowId = "")
			s arrData = GetCodeTypeArr(defFlag, scdRowId)
			s code = $lg(arrData, 1)
			s desc = $lg(arrData, 2)
			s rowObj = ##class(PHA.COM.Object).%New()
			d rowObj.%Set("RowId", code)
			d rowObj.%Set("Description", desc)
			d retArr.%Push(rowObj)
		}
	}
	k GetCodeTypeArr
	q retArr.%ToJSON()
}

/// Description:通过追溯码获取对应药品信息
/// Creator:    hulihua
/// CreateDate: 2024-09-03
/// Table:      
/// Input:      TraceCode-追溯码、CodeTypeId-码类型ID
/// Output:        
/// Return：    追溯码对应药品信息
/// Others:     
/// Debug:      w ##class(PHA.DTC.Scan.Query).GetDrugByCode("11811373", "27")
ClassMethod GetDrugByCode(TraceCode, CodeTypeId) As %String
{
	q:(CodeTypeId = "") "-1^码类型不能为空！"
	q:(TraceCode = "") "-1^追溯码为空！"
	i CodeTypeId [ "TC" {
        s CodeTypeId = $o(^DHCSTSCDI(0,"TYPECODE",$zcvt("CodeType","U"),CodeTypeId,""))
    }
	s typeCode = $p($g(^DHCSTSCDI(+CodeTypeId)), "^", 1)
	s typeDesc = $p($g(^DHCSTSCDI(+CodeTypeId)), "^", 2)
	if (($e(TraceCode, 1, 2) = "82") || ($e(TraceCode, 1, 2) = "89")) {
		s idCode = $e(TraceCode, 1, 8)
	} else {
		s idCode = $e(TraceCode, 1, 7)
	}
	s inciId = $o(^CT.PHA.IN.IncIDCodeI("TypeActCodeInc", CodeTypeId, 1, idCode, ""))
	q:(inciId = "") "-1^该药品对应的标识码未关联到药品字典！"
	s existFlag = ##class(Save).IsSplit(typeCode, TraceCode)
	q:(existFlag) "-1^该追溯码已拆零！"
	s inciCode = $p(^INCI(inciId,1),"^",1)
	s inciDesc = $p(^INCI(inciId,1),"^",2)
	s inciSpec = ##class(web.DHCST.Common.DrugInfoCommon).GetSpec("", inciId)
	s manfName = $p(##class(web.DHCST.Common.DrugInfoCommon).GetManf(inciId), "^", 3)
	s bUomId = $p(^INCI(inciId,1),"^",10)
	s pUomId = $p(^INCI(inciId,1), "^", 17)
	if (pUomId = "") {
		s pUomId = $p(^INCI(inciId,3), "^", 6)
	}
	s inciFac = ##class(PHA.DTC.COM.Method).UomFac(pUomId,bUomId)
    s packFac = 1
    if (typeCode = "BTC") {
        s packUomStr = ##class(web.DHCST.Common.DrugInfoCommon).GetIncPackUom(inciId)
        s packFac = +$p(packUomStr, "^", 3)
        if (packFac = 0) {
            s packFac = 1
        }
    } elseif (typeCode = "MTC") {
        s packUomMStr = ##class(web.DHCST.Common.DrugInfoCommon).GetIncPackUomM(inciId)
        s packFac = +$p(packUomMStr, "^", 3)
        if (packFac = 0) {
            s packFac = 1
        }
    }
    s barQty = inciFac * packFac
	s bUomDesc = $p($g(^CT("UOM",+bUomId)),"^",2)
    s codeObj = ##class(PHA.COM.Object).%New()
    d codeObj.%Set("inciId", inciId)
    d codeObj.%Set("idCode", idCode)
    d codeObj.%Set("typeDesc", typeDesc)
    d codeObj.%Set("traceCode", TraceCode)
    d codeObj.%Set("typeDr", CodeTypeId)
    d codeObj.%Set("inciCode", inciCode)
    d codeObj.%Set("inciDesc", inciDesc)
    d codeObj.%Set("inciSpec", inciSpec)
    d codeObj.%Set("manfName", manfName)
    d codeObj.%Set("barQty", barQty)
    d codeObj.%Set("uomDesc", bUomDesc)
    d codeObj.%Set("uomId", bUomId)
    q codeObj.%ToJSON()
}
