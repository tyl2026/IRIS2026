//工具函数-以HTML的形式显示扫描状态

/// 查询
function Query() {
    var startDate = $('#StartDate').datebox('getValue');
    var endDate = $('#EndDate').datebox('getValue');
    var methType = $('#cmbMethType').combobox('getValue');
    var chFlag = $('#cmbChFlag').combobox('getValue');
    var deleteType = $('#cmbDeleteType').combobox('getValue');
    var batchno = $('#txtbatchno').val().trim();
    var hospId = $('#cmbHospital').combobox('getValue');
    var typeManf = $('#cmbTypeManf').combobox('getValue');
    var selType = $('#cmbSelType').combobox('getValue') || '';
    var admType = $('#cmbAdmType').combobox('getValue') || '';
    var scanStatus = $('#cmbScanStatus').combobox('getValue') || '';
    var phaLocId = $('#cmbPhaLocId').combobox('getValue') || '';
    var incidesc = $.trim($('#cmbgridInci').lookup('getText')) || '';
    var patNo = $('#txtPatNo').val();
    var traceCode = $('#txtTraceCode').val();
    var dateType = $('#cmbDateType').combobox('getValue');
    var params = startDate + '^' + endDate + '^' + methType + '^' + chFlag + '^' + deleteType + '^' +
        batchno + '^' + hospId + '^' + typeManf + '^' + selType + '^' + admType + '^' + scanStatus + '^' +
        phaLocId + '^' + incidesc + '^' + patNo + '^' + traceCode + '^' + dateType;
    $('#gridInsuData').datagrid('query', {
        inputStr: params
    });
}

/// 补传
function Save() {
    var gridLogChecked = $("#gridInsuData").datagrid("getChecked") || "";
    if (gridLogChecked == "") {
        $.messager.alert("温馨提示", "请勾选需要补传的数据", 'info');
        return;
    }
    $.messager.progress({
        title: "提示",
        msg: '正在补传数据',
        text: '补传中....'
    });
    setTimeout(function () {
        var warnType = "";
        for (var i = 0; i < gridLogChecked.length; i++) {
            var checkedData = gridLogChecked[i];
            var checkFlag = checkedData.CheckFlag;
            if (checkFlag === "Y") {
                warnType = "1";
                continue;
            }
            var pOutRowId = checkedData.Pointer;
            var pType = checkedData.Type;
            var pHospId = checkedData.HospId;
            var pTrdnFlag = checkedData.TrdnFlag;
            var pInsuManf = checkedData.InsuManfCode;
            var options = {
                ClassName: "PHA.FACE.TPS.INSUTE.LogQuery",
                MethodName: "SendRepairData",
                FaceType: pType,
                Pointer: pOutRowId,
                HospId: pHospId,
                TrdnFlag: pTrdnFlag,
                InsuManf: pInsuManf
            }
            var ret = $m(options, false);
        }
        $.messager.progress("close");
        if (warnType === "1") {
            $.messager.alert("温馨提示", "已扫码且已经上传成功的数据不会再次补传", 'info');
        }
        Query();
    }, 200);
}

/// 删除
function Delete() {
    var gridLogChecked = $("#gridInsuData").datagrid("getChecked") || "";
    if (gridLogChecked == "") {
        $.messager.alert("温馨提示", "请请勾选需要删除的数据", 'info');
        return;
    }
    var warnType = "";
    var batchArr = [];
    for (var i = 0; i < gridLogChecked.length; i++) {
        var checkedData = gridLogChecked[i];
        var CheckFlag = checkedData.CheckFlag;
        if (CheckFlag === "N") {
            warnType = "1";
            continue;
        }
        var Batch = checkedData.Batch;
        var DeleteType = checkedData.DeleteType;
        var InsuManf = checkedData.InsuManfCode;
        var batchStr = Batch + "^" + DeleteType + "^" + InsuManf;
        batchArr.push(batchStr);
    }
    if (warnType === "1") {
        $.messager.alert("温馨提示", "上传失败的数据是不允许删除的", 'info');
    }
    var hospId = $('#cmbHospital').combobox('getValue');
    var options = {
        ClassName: "PHA.FACE.TPS.INSUTE.Upload",
        MethodName: "DeleteData",
        BatchStr: batchArr.join(","),
        HospId: hospId
    }
    var ret = $m(options, false);
    Query();
}

function setInputInfoFormatter(value, row, index) {
    var nfhml = "<span class='linkinfo' onclick='showInputJson(" + index + ")'>入参详情</span>";
    return nfhml;
}

function setOutputInfoFormatter(value, row, index) {
    var nfhml = "<span class='linkinfo' onclick='showOutJson(" + index + ")'>出参详情</span>";
    return nfhml;
}

function showInputJson(index) {
    $("#info").html("");
    var rowData = $('#gridInsuData').datagrid('getRows')[index];
    var logId = rowData.RowId;
    var value = tkMakeServerCall('PHA.FACE.TPS.INSUTE.LogQuery', 'GetLogInText', logId);
    try {
        var ret = JSON.stringify(JSON.parse(value), null, 4);
    } catch (ex) {
        var ret = rowData.Text;
    }
    $("#info").html("<pre id='copyInfo'>" + ret + "</pre>");
    $('#InfoWin').dialog("setTitle", "入参");
    $('#InfoWin').dialog("open");
}

function showOutJson(index) {
    $("#info").html("");
    var rowData = $('#gridInsuData').datagrid('getRows')[index];
    var value = rowData.OutText;
    try {
        var ret = JSON.stringify(JSON.parse(value), null, 4);
    } catch (ex) {
        var ret = value;
    }
    $("#info").html("<pre id='copyInfo'>" + ret + "</pre>");
    $('#InfoWin').dialog("setTitle", "出参");
    $('#InfoWin').dialog("open");
}

/// 3505的删除 , 得调用3506A
function Delete3505() {
	var gridLogChecked = $("#gridInsuData").datagrid("getChecked") || "";
    if (gridLogChecked == "") {
        $.messager.alert("温馨提示", "请请勾选需要删除的数据", 'info');
        return;
    }
    var warnType = "";
    var logIdArr = [];
    for (var i = 0; i < gridLogChecked.length; i++) {
        var checkedData = gridLogChecked[i];
        var CheckFlag = checkedData.CheckFlag;
        if (CheckFlag === "N") {
            warnType = "1";
            continue;
        }
        var logType = checkedData.Type;
        if ((logType != 'Method3505')&&(logType != 'Method3505A')) {
	    	warnType = "1";
            continue;
	    }
        var logId = checkedData.RowId;
        logIdArr.push(logId);
    }
    if (warnType === "1") {
        $.messager.alert("温馨提示", "上传失败或非销售数据无需删除", 'info');
    }
    $.messager.confirm('操作确认', '您确认删除当前选择的已上传销售数据吗？', function(r) {
    	if (r) {
        	var hospId = $('#cmbHospital').combobox('getValue');
		    var options = {
		        ClassName: "PHA.FACE.TPS.INSUTE.Upload",
		        MethodName: "DeleteBy3506",
		        LogStr: logIdArr.join(","),
		        HospId: hospId,
		        UserId: session['LOGON.USERID']
		    }
		    var ret = $m(options, false);
		    Query();
    	}else {
	    	Query();
	    }
	});
}
