/// Creator：    徐保保
/// CreatDate：  2021-12-29
/// Description:  4101-医疗保障基金结算清单信息上传(setlinfo)
/// Input：		 AdmDr 			就诊号
/// 				 ExpDataAry		共同数组
/// 			 PrtDr 			发票表id
/// 			 PBDDr			医保登记表id
/// Output：	 4201 文档结果输出
/// Others：	 其它说明
/// w ##class(web.DHCINSUSetlInfo).GetSetlInfo(128245,"",55222)
ClassMethod GetSetlInfo(AdmDr As %String, InsuAdmDr As %String, InsuDivDr As %String) As %ArrayOfDataTypes
{
	
	s setlinfo=##class(%ArrayOfDataTypes).%New()
	q:InsuDivDr="" setlinfo
	s:AdmDr="" AdmDr=$p($g(^DHCINDIV(InsuDivDr)),"^",1)
	q:AdmDr="" setlinfo
	
	
	s billDr=$p($g(^DHCINDIV(InsuDivDr)),"^",3)
	s PrtDr=$p($g(^DHCINDIV(InsuDivDr)),"^",4)
	s admType=$p(^PAADM(AdmDr),"^",2)			//就诊类型
	i admType="I" d
	.q:billDr=""
	.s PrtDr=$o(^DHCINVPRTZY(0,"AR",billDr,""),-1)  
	q:PrtDr="" setlinfo							//未财务结算退出
	
	//获取共同参数数组信息
	s ExpDataAry=""
	s rtnFlag=..GetInsuReportInfoCom(AdmDr,.ExpDataAry,PrtDr,InsuDivDr)
	q:rtnFlag'="0" setlinfo
	//将共同参数数据赋给结算清单数组
	if ($IsObject(ExpDataAry)){
		f i=1:1:ExpDataAry.Count() {
			do ExpDataAry.GetNext(.key)
			s value=ExpDataAry.GetAt(key)
			d setlinfo.SetAt(value,key)
		}
    }
    
    
	s insuDate=$p($g(^DHCINDIV(InsuDivDr)),"^",16)
	s insuTime=$p($g(^DHCINDIV(InsuDivDr)),"^",17)
	s dclatime=$zd(insuDate,3)_" "_$zt(insuTime) 
	d setlinfo.SetAt(dclatime,"dcla_time")				;结算清单上报时间 
	
	
	//1.方式一：取病案首页上的信息，证据类型写死取身份证证件号码取病案首页身份证号
	//2.方式二：医生站提供接口，取HIS中的证件类型和证件号
	//标准版使用方式一，项目根据情况跟医院确认
	s patncerttype="1"		
	d setlinfo.SetAt(patncerttype,"patn_cert_type")		;患者证件类别
	s insuplc=setlinfo.GetAt("insu_admdvs")			
	d setlinfo.SetAt(insuplc,"insuplc")					;参保地
	s sppsntype=""
	d setlinfo.SetAt(sppsntype,"sp_psn_type")			;特殊人员类型(1:特困人员,2:城乡低保对象,3:农村建档立卡贫困人口,4:贫困重度残疾人,9:其他)
	s opspdiagcaty=""
	d setlinfo.SetAt(opspdiagcaty,"opsp_diag_caty")		;门诊慢特病诊断科别
	s opspmdtrtdate=$p(^PAADM(AdmDr),"^",6)
	s:opspmdtrtdate'="" opspmdtrtdate=$zd(opspmdtrtdate,3)
	d setlinfo.SetAt(opspmdtrtdate,"opsp_mdtrt_date")	;门诊慢特病就诊日期
	
	b ;1
	
	//获取电子病历术语集数据H开头
	s EMSetlInfo=##class(%ArrayOfDataTypes).%New()
	s EMSetlInfo=##class(EMRservice.BL.Interface.RecordInfoToINSU).setlinfo4101(AdmDr)
	b ;00
	if ($IsObject(EMSetlInfo)){
		f i=1:1:EMSetlInfo.Count() {
			do EMSetlInfo.GetNext(.emkey)
			s value=EMSetlInfo.GetAt(emkey)
			d setlinfo.SetAt(value,"H"_emkey)
		}
    }
    b ;2
	
	//获取病案编目数据P开头
	s rset=##class(%ResultSet).%New("MA.IPMR.IO.Insurance.For4101:Qrysetinfo")
	d rset.Execute(AdmDr)
	s columns = rset.GetColumnCount()
	f i=1:1:columns d
	.//s Columnkey=rset.ColIndex(i)
	.s Columnkey=rset.GetColumnName(i)
	.s value=rset.GetDataByName(Columnkey)
	.d setlinfo.SetAt(value,"P"_Columnkey)
	b ;3
	
    //其他产品组接口---------------
	
	q setlinfo
}

/// Output：	 医疗收费项目,金额,甲类费用合计,乙类金额,全自费金额,其他金额
/// Others：	 其它说明

Query PatFeeItemInfoQuery(AdmDr As %String, InsuAdmDr As %String, InsuDivDr As %String) As websys.Query(ROWSPEC = "med_chrgitm:%String,amt:%Double,claa_sumfee:%Double,clab_amt:%Double,fulamt_ownpay_amt:%Double,oth_amt:%Double") [ SqlProc ]
{
}

ClassMethod PatFeeItemInfoQueryExecute(ByRef qHandle As %Binary, AdmDr As %String, InsuAdmDr As %String, InsuDivDr As %String) As %Status
{
	s repid=$I(^CacheTemp)
	s ind=1
	s qHandle=$lb(0,repid,0)
	
	if (InsuDivDr="") Set qHandle=$lb(0,repid,0) Quit $$$OK
	
	k Sort("InsuIteminfo",InsuDivDr)
	
    if '$d(^DHCINDIS("0","DivideDr",InsuDivDr))	d
    .s PBDr=$p($g(^DHCINDIV(InsuDivDr)),"^",3)		//账单号
    .i PBDr="" d
    ..s DhcInvPrtDr=$p($g(^DHCINDIV(InsuDivDr)),"^",3)	//门诊发票ID
    ..q:'$d(^DHCBCI(0,"INV",DhcInvPrtDr)) 
    ..s BCInvDr=""
	..f  s BCInvDr=$o(^DHCBCI(0,"INV",DhcInvPrtDr,BCInvDr)) q:BCInvDr=""  d
	...s PBDr=$p(^DHCBCI(BCInvDr),"^",2)  				;账单号
	...q:PBDr=""
	...d GetDivSubIDByPBDr(PBDr)
	.d GetDivSubIDByPBDr(PBDr)
	e  d
	.d GetDivSubIDByDivDr(InsuDivDr) 
	
	s (AmtSum,ClaaSumfeeSum,ClabAmtSum,OwnpayAmtSum,OthAmtSum)=0
	s MedChrgitm=""
	f  s MedChrgitm=$o(Sort("InsuIteminfo",InsuDivDr,MedChrgitm)) q:MedChrgitm=""  d
	.s AmtSum=$p(Sort("InsuIteminfo",InsuDivDr,MedChrgitm),"^",2)
	.s ClaaSumfeeSum=$p(Sort("InsuIteminfo",InsuDivDr,MedChrgitm),"^",3)
	.s ClabAmtSum=$p(Sort("InsuIteminfo",InsuDivDr,MedChrgitm),"^",4)
	.s OwnpayAmtSum=$p(Sort("InsuIteminfo",InsuDivDr,MedChrgitm),"^",5)
	.s OthAmtSum=$p(Sort("InsuIteminfo",InsuDivDr,MedChrgitm),"^",6)
    .d BuildFeeItemInfo
    
    Quit $$$OK
    
    //通过账单获取费用明细上传数据
GetDivSubIDByPBDr(PBDr)
	s InsuDivSubID=""
	f  s InsuDivSubID=$o(^DHCINDIS("0","PBDr",PBDr,InsuDivSubID)) q:InsuDivSubID=""  d	
	.d GetFeeItemInfo(InsuDivSubID)
	
	//通过医保结算表id获取费用明细上传数据
GetDivSubIDByDivDr(InsuDivID)	
	s InsuDivSubID=""
    f  s InsuDivSubID=$o(^DHCINDIS("0","DivideDr",InsuDivDr,InsuDivSubID)) q:InsuDivSubID=""  d
    .d GetFeeItemInfo(InsuDivSubID)
    
GetFeeItemInfo(InsuDivSubID)
	
	//其他金额默认，需根据项目核实确定
	s (ClaaSumfee,ClabAmt,OwnpayAmt,OthAmt)=0
	i InsuDivSubID=""  b ;12
	s InsuDivSubInfo=$g(^DHCINDIS(InsuDivSubID))
	q:InsuDivSubInfo=""
	s Flag=$p(InsuDivSubInfo,"^",18)
	q:Flag'="N"   //未上传数据不统计      ---此处根据项目情况暂定
	s INDISINSUXMLB=$p(InsuDivSubInfo,"^",10)			;甲乙丙
	s INDISAmount=+$p(InsuDivSubInfo,"^",13)
	s INDISScale=$p(InsuDivSubInfo,"^",15)	;自付比例
	s INDISFund=$p(InsuDivSubInfo,"^",16)
	s INDISSelf=+$p(InsuDivSubInfo,"^",17)	;先行自付
	s INDISDemo1=+$p(InsuDivSubInfo,"^",26)	;全自费
	s INDISDemo2=+$p(InsuDivSubInfo,"^",27)	;符合范围
	s INDISDemo3=+$p(InsuDivSubInfo,"^",28)	;超限
	s INDISDemo4=$p(InsuDivSubInfo,"^",29)	;医疗收费项目   ;根据当地当地程序修改
    i (INDISINSUXMLB="01")||(INDISINSUXMLB="甲类") d
	.s ClaaSumfee=INDISAmount	; 甲类费用        
	e  i (INDISINSUXMLB="02")||(INDISINSUXMLB="乙类") d
	.s ClabAmt=INDISAmount	    ; 乙类费用        
	e   d
	.s OwnpayAmt=INDISAmount    ;全自费金额                    
    ;s OwnpayAmt=OwnpayAmt+INDISDemo3	;全自费金额=全自费金额+超限价金额
    
    i $d(Sort("InsuIteminfo",InsuDivDr,INDISDemo4))=0 d
    .s Sort("InsuIteminfo",InsuDivDr,INDISDemo4)=INDISDemo4_"^"_INDISAmount_"^"_ClaaSumfee_"^"_ClabAmt_"^"_OwnpayAmt_"^"_OthAmt
	e  d
	.s $p(Sort("InsuIteminfo",InsuDivDr,INDISDemo4),"^",2)=+$p(Sort("InsuIteminfo",InsuDivDr,INDISDemo4),"^",2)+INDISAmount
	.s $p(Sort("InsuIteminfo",InsuDivDr,INDISDemo4),"^",3)=+$p(Sort("InsuIteminfo",InsuDivDr,INDISDemo4),"^",3)+ClaaSumfee
	.s $p(Sort("InsuIteminfo",InsuDivDr,INDISDemo4),"^",4)=+$p(Sort("InsuIteminfo",InsuDivDr,INDISDemo4),"^",4)+ClabAmt
	.s $p(Sort("InsuIteminfo",InsuDivDr,INDISDemo4),"^",5)=+$p(Sort("InsuIteminfo",InsuDivDr,INDISDemo4),"^",5)+OwnpayAmt
	.s $p(Sort("InsuIteminfo",InsuDivDr,INDISDemo4),"^",6)=+$p(Sort("InsuIteminfo",InsuDivDr,INDISDemo4),"^",6)+OthAmt
	
    Quit $$$OK
BuildFeeItemInfo	
	set Data=$lb(MedChrgitm,AmtSum,ClaaSumfeeSum,ClabAmtSum,OwnpayAmtSum,OthAmtSum)
 	Set ^CacheTemp(repid,ind)=Data
 	Set ind=ind+1
 	q
}

	}

	s OutArgType="JSON"

		s Ret=##class(INSU.OFFBIZ.BL.BIZ00A).InsuDocDetailedAuditService(Param,dataInputArgArr)

		s Ret=##class(INSU.OFFBIZ.BL.BIZ00A).InsuDocDetailedAuditService(Param,dataInputArgArr)

		d dataInputArgArr.SetAt(DataObj,"data")	
		s Ret=##class(INSU.OFFBIZ.BL.BIZ00A).InsuDocDetailedAuditService(Param,dataInputArgArr)
	}

/// 根据医嘱id判断是否为医保结算（药房药库调用）
/// 入参：  医嘱id
/// 出参： Y:医保,N:非医保
/// 使用场景：根据医嘱id来判断该医嘱是否为医保结算。入参门诊为医嘱id OEORIId、住院为医嘱执行表id OEOREId，出参为医保标志
/// +靳帅1010 20221209 upt 靳帅1010 20230404
/// 门诊-OE_Order-Pa_Adm-OE_OrdItem-DHC_PatientBill-DHC_BillConINV-DHC_INVPRT-INSU_Divide-判断：医保结算状态I为医保结算
/// 住院-OE_Order-Pa_Adm-OE_OrdExec-DHC_PatientBill-INSU_Divide-判断：医保结算状态I为医保结算
/// w ##class(INSU.BL.Interface.DataRpUp).GetIsInsuPayByOEORDId("7||4","17||1||1") 
ClassMethod GetIsInsuPayByOEORDId(OEORIId As %String, OEOREId As %String) As %String
{
   s $zt="GetIsInsuPayByOEORDIdEx"
   q:(OEORIId="")&&(OEOREId="") "-1^医嘱ID和执行记录ID不能同时为空"
   s InsuFlag="N"     
   s OEOrdRowId=$P(OEORIId,"||",1)
   s:OEOrdRowId="" OEOrdRowId=$P(OEOREId,"||",1)
   s PaadmDr=$p($g(^OEORD(OEOrdRowId)),"^",1) //就诊Dr
   s PaadmInfo=$g(^PAADM(PaadmDr))
   s PaadmType=$p(PaadmInfo,"^",2)   //就诊类型
   //如果就诊类型不为"I"(住院)或"P"(预住院) 为门诊医嘱
   i (PaadmType'="I")&&(PaadmType'="P")
      {
          d OpGetIsInsuPay   //门诊医嘱查询该医嘱对应的医保结算状态
      }                
   else
      {                           
          d IPGetIsInsuPay   //住院医嘱查询医保结算状态
      }
   q InsuFlag
OpGetIsInsuPay
   s PatBillId=""
   s PatBillId=$o(^DHCPBi(0,"OEORI",OEORIId,PatBillId),-1)
   q:PatBillId="" 
   s PbInfo=$g(^DHCPB(PatBillId))
   s BillConINVId="" 
   s BillConINVId=$o(^DHCBCI(0,"Bill",PatBillId,BillConINVId),-1)
   q:BillConINVId=""  
   s BillConINVInfo=$g(^DHCBCI(BillConINVId))
   s DHCINVPRTDr=$p(BillConINVInfo,"^",1)  //发票表ID
   s DivideDr=""
   s DivideDr=$O(^DHCINDIV("0","DHCInvPrt",DHCINVPRTDr,DivideDr),-1)
   //当根据索引，取到DivideDr为空时,不为医保结算，退出
   q:DivideDr=""
   s DivideInfo=$g(^DHCINDIV(DivideDr))
   s INPAYFlag=$p(DivideInfo,"^",5)    //医保结算状态
   //当医保结算状态未为"I"(正常)时，正常结算数据
   i (INPAYFlag="I")   
        {
            s InsuFlag="Y" 
        }          
   q
IPGetIsInsuPay
   s IpPatBillId=""
   s IpPatBillId=$o(^DHCPB(0,"OEEXC",OEOREId,IpPatBillId),-1) 
   q:IpPatBillId="" 
   s IpPbInfo=$g(^DHCPB(IpPatBillId))
   s IPPayedFlag=$p(IpPbInfo,"^",16)      //计费状态(Paid,Bill)
   //当账单计费状态为"P"时是已经结算账单
   i (IPPayedFlag="P")
     {
        s DivideDr=""
        s DivideDr=$O(^DHCINDIV("0","DHCPB",IpPatBillId,DivideDr),-1)
        //当根据索引，取到DivideDr为空时，不为医保结算，退出
        q:DivideDr=""
        s DivideInfo=$g(^DHCINDIV(DivideDr))
        s INPAYFlag=$p(DivideInfo,"^",5)    //医保结算状态
        //当医保结算状态未为"I"时，正常结算数据
        i (INPAYFlag="I")  
            {
                s InsuFlag="Y"
            }                
     }
   q
GetIsInsuPayByOEORDIdEx
   s $zt=""
   q "-99^程序异常："_$ze
}
