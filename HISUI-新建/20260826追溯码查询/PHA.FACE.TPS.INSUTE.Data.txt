/// Description: 拉取3512方法,将3512的返回中的配送批次流水
/// Creator: 	 tanwenbin
/// CreatDate:	 20250812
/// Table:		
/// Input:		 inci 库存项ID, traceCode追溯码, UserId拉取的用户ID, HospId 院区ID, InsuManf
/// Output:		 
/// Return:		 成功：fixmedinsBchno，失败："" 原因查日志表 BS_PHA_IN.InsuDataLog
/// Others:
/// w ##class(PHA.FACE.TPS.INSUTE.Data).get3512FixmedinsBchno(5175,"83830471090549422714",1,"2","1")
ClassMethod get3512FixmedinsBchno(inciId, traceCode, UserId, HospId, InsuManf)
{
	q:traceCode="" ""
	s titleObj = ##class(PHA.COM.Object).%New()
	d ##class(Format).SetTitleData(titleObj, UserId, HospId, InsuManf)
	d titleObj.%Set("infno", "3512")
	s inciCode = $p(^INCI(inciId,1),"^",1)
	s insuCode = $p(..GetInsuData(inciId, HospId), "^", 1)
	s busData = $lb(insuCode,inciCode,traceCode)
	s busDetObj = ##class(Format).DataToJSON("3512", busData, InsuManf, HospId)
	s inputObj = ##class(PHA.COM.Object).%New()
	d inputObj.%Set("data", busDetObj)
	d titleObj.%Set("input", inputObj)
	s logStr = $lb(traceCode,"",inciId,0)
	s insertRet = ##class(LogSave).SaveLog("Method3512", (titleObj.%ToJSON()), logStr, HospId, InsuManf)

	s retObj = ..SendToInsuFace(titleObj, HospId, InsuManf)

	s infoCode = retObj.infcode
	s fixmedinsBchno = "" ;retObj.output.%Get(0).fixmedinsBchno
	s retInfo = retObj.%ToJSON()
	s pointer = traceCode
	s trdnFlag = 0
	if (infoCode '= 0) {
		d ##class(LogSave).UpdateLog("Method3512", retInfo, pointer, trdnFlag, "N", InsuManf)
	} else {
		s:retObj.output'="null" fixmedinsBchno = retObj.output.%Get(0).fixmedinsBchno
		d ##class(LogSave).UpdateLog("Method3512", retInfo, pointer, trdnFlag, "Y", InsuManf)
	}
	d inputObj.%Clear()
	d retObj.%Clear()
	d titleObj.%Remove("input")
	q fixmedinsBchno
}

/// Description: 获取3501商品盘存上传接口所需数据
/// Creator: 	 hulihua
/// CreatDate:	 2024-12-12
/// Table:		
/// Input:		 Date-数据日期, BusId-业务表ID, InstItmId-盘点明细表ID, HospId-院区ID, TrdnFlag-拆零标志, InsuManf-医保厂商, Pid-计数器, TaskFlag-系统任务标志
/// Output:		 
/// Return:		 成功：0，失败：-1^失败原因
/// Others:
/// Debug:		 w ##class(PHA.FACE.TPS.INSUTE.Data).Get3501Data()
ClassMethod Get3501Data(Date, BusIdStr, InstItmId, HospId, TrdnFlag, InsuManf, Pid, TaskFlag)
{
	q:(..CheckHosp(HospId) '= 1) ..Error(-1, "该接口未授权，请联系系统管理员！")
	q:(Date = "")&&(BusIdStr = "")&&(InstItmId = "") ..Error(-1, "日期、业务或盘点明细ID不能同时为空！")
	d ..ClearTmp("PHA.FACE.TPS.INSUTE.Data", "Get3501Data", Pid)
	s paramStr = "^^^"_HospId
	s configStr = ..GetParamProp(paramStr)
	s uomType = $p(configStr, "^", 1)
	s resQtyType = $p(configStr, "^", 2)
	s upExtTCodeFlag = $p(configStr, "^", 5)
	s ret = 0
	if (InstItmId '= "") {
		q:('$d(^DHCINST(+InstItmId, "STI", $p(InstItmId, "||", 2)))) ..Error(-1, "该明细在盘点表中不存在！")
		s locId = $p($g(^DHCINST(+InstItmId)), "^", 5)
		s hospId = $p(^CTLOC(locId), "^", 22)
		q:((hospId '= "") && (HospId '= hospId)) ..Error(-1, "该台帐院区不匹配！")
		s instItmId = InstItmId
		d BuildData
	} elseif (BusIdStr '= "") {
		s instId = $p(BusIdStr, "^", 1)
		q:('$d(^DHCINST(instId))) ..Error(-1, "该盘点单数据在系统中不存在！")
		s locId = $p(^DHCINST(instId), "^", 5)
		s hospId = $p(^CTLOC(locId), "^", 22)
		q:((hospId '= "") && (HospId '= hospId)) ..Error(-1, "该盘点单院区不匹配！")
		s instCh = ""
		for {
			s instCh = $o(^DHCINST(instId, "STI",instCh))
			q:(instCh = "")
			continue:('$d(^DHCINST(instId, "STI",instCh)))
			s instItmId = instId_"||"_instCh
			d BuildData
		}
	} else {
		s instId = ""
		for {
			s instId = $o(^DHCINST(0, "DATE", Date, instId))
			q:(instId = "")
			continue:('$d(^DHCINST(instId)))
			s locId = $p(^DHCINST(instId), "^", 5)
			s hospId = $p(^CTLOC(locId), "^", 22)
			continue:((hospId '= "") && (HospId '= hospId))
			s instCh = 0
			for {
				s instCh = $o(^DHCINST(instId, "STI",instCh))
				q:(instCh = "")
				continue:('$d(^DHCINST(instId, "STI",instCh)))
				s instItmId = instId_"||"_instCh
				d BuildData
			}
		}
	}
	q ret
BuildData
	s instItmData = $g(^DHCINST(+instItmId, "STI", $p(instItmId, "||", 2)))
	s inclb = $p(instItmData, "^", 1)
	s inciId = +inclb
	s stkGrpStr = ..GetStkCatGrp(inciId)
	/* 过滤-非药品 */
	q:($p(stkGrpStr, "^", 3) '= "G") ""
	/* 过滤-已上传 */
	s indexId = instItmId
	q:(..GetSuccFlag("Method3501",indexId, InsuManf, TrdnFlag) = "Y") ""
	s insuCode = $p(..GetInsuData(inciId, HospId), "^", 1)
	s inciCode = $p(^INCI(inciId, 1), "^", 1)
	s inciDesc = $p(^INCI(inciId, 1), "^", 2)
	s rxFlag = ..GetInciOtcFlag(inciId)
	s instDate = $p(^DHCINST(+instItmId), "^", 5)
	if (instDate '= "") {
		s instDate = $zd(instDate, 3)
	}
	s bchNo = instItmId
	s batData = ..GetBatByInclb(inclb)
	s batNo = $lg(batData, 1)
	s expDate = $lg(batData, 2) 
	s produceDate = $lg(batData, 3)
	s remarks = ""													// 备注
	s countQty = $p(instItmData, "^", 5)
	s instUomId = $p(instItmData, "^", 17)
	s bUomId = $p(^INCI(inciId, 1), "^", 10)
 	s instFac = ..UomFac(instUomId, bUomId)
 	s bCountQty= countQty * instFac
 	s pUomId = $p(^INCI(inciId, 1), "^", 17)
 	if (pUomId = "") {
		s pUomId = $p(^INCI(inciId, 3), "^", 6)
 	}
 	s inciFac = ..UomFac(pUomId, bUomId)
	s retCodeObj = ..GetTraceCode(instItmId)
	s pQty = retCodeObj.%Get("PQty")
	s pTCodeArr = retCodeObj.%Get("PTCode")
	if (pTCodeArr.%Size() > 0) {
		s trdnFlag = 0
		if ((uomType = "1") || (uomType = "2")) {
			s qty = pQty / inciFac
		} else {
			s qty = pQty
		}
		s traceCodeArr = pTCodeArr
		d OutPutRow
	}
	s sTCodeArr = retCodeObj.%Get("STCode")
	if (resQtyType = "1") {
		s sQty = retCodeObj.%Get("SQty")
	} else {
		s sQty = bCountQty - pQty
	}
	if (sQty > 0) {
		s trdnFlag = 1
		if (uomType = "1") {
			s qty = sQty / inciFac
		} else {
			s qty = sQty
		}
		s traceCodeArr = sTCodeArr
		d OutPutRow
	} elseif (pTCodeArr.%Size() = 0) {
		s sQty = bCountQty
		s trdnFlag = 1
		if (uomType = "1") {
			s qty = sQty / inciFac
		} else {
			s qty = sQty
		}
		s traceCodeArr = sTCodeArr
		d OutPutRow 
	}
	q 0
 
OutPutRow
	q:(TrdnFlag '= "")&&(TrdnFlag '= trdnFlag)
	s noTCodeFlag = retCodeObj.%Get("NoTCodeFlag")
	if (traceCodeArr.%Size() > 0) {
		if (noTCodeFlag = "N") {
			s existCode = "Y"
		} else {
			s existCode = "C"
		}
	} else {
		s existCode = "N"
		q:(TaskFlag = "N")||(upExtTCodeFlag = "Y")
	}
	s tCodeJsonStr = traceCodeArr.%ToJSON()
	s data1 = $lb(insuCode, inciCode, inciDesc, rxFlag, instDate)
	s data2 = $lb(qty, batNo, bchNo, produceDate, expDate)
	s data3 = $lb(remarks, tCodeJsonStr)
	s data = data1_data2_data3
	s ^||TMP("PHA", "PHA.FACE.TPS.INSUTE.Data", "Get3501Data", Pid, "Data", +instItmId, $p(instItmId, "||", 2), trdnFlag) = data
	s logData = $lb(indexId, bchNo, inciId, trdnFlag, existCode)
	s ^||TMP("PHA", "PHA.FACE.TPS.INSUTE.Data", "Get3501Data", Pid, "Log", +instItmId, $p(instItmId, "||", 2), trdnFlag) = logData
	q
}

			s locId = $p(^DHCINGR(ingdRecId), "^", 13)

			q:(hospId '= "")&&(HospId '= hospId) ..Error(-1, "该入库单院区不匹配！")

					q:(intrId = "")
					continue:('$d(^DHCINTR(intrId)))
					s inclb = $p(^DHCINTR(intrId), "^", 7)
					d BuildData
				}
			}
		} elseif (busType = "D") {

BuildData
	s inciId = +inclb

	s bchNo =..getApprovalNoByIncib(incib) //incib  // intrId
	i bchNo="" s bchNo=incib
	s intrTime = $p(intrData, "^", 3)
	if (intrDate '= "") {
		s chgDate = $zd(intrDate, 3)
	}
	if (intrTime '= "") {
		s chgTime = $zt(intrTime, 1)
	}
	s chgDateTime = chgDate_" "_chgTime
	s intrUserId = $p(intrData, "^", 11)
	s intrUserStr = ..GetPharUserStr(intrUserId)
	s intrUserName = $p(intrUserStr, "^", 1)
	s remarks = ""													// 备注
	s intrUom = $p(intrData, "^", 10)
	s intrQty = $ZABS($p(intrData, "^", 6))
	s intrSp = +$p(intrData, "^", 14)
	s bUomId = $p(^INCI(inciId, 1), "^", 10)
	s pUomId = $p(^INCI(inciId, 1), "^", 17)

 	}
 	s intrFac = ..UomFac(intrUom, bUomId)
 	s inciFac = ..UomFac(pUomId, bUomId)
	s bQty = intrQty * intrFac
	s bSp = intrSp / intrFac
	s retCodeObj = ..GetTraceCode(intrId)
	s pQty = retCodeObj.%Get("PQty")
	s pTCodeArr = retCodeObj.%Get("PTCode")
	if (pTCodeArr.%Size() > 0) {
		s trdnFlag = 0
		if ((uomType = "1") || (uomType = "2")) {
			s qty = pQty / inciFac
			s sp = bSp * inciFac
		} else {
			s qty = pQty
			s sp = bSp
		}

	}

	}

		s trdnFlag = 1
		if (uomType = "1") {
			s qty = sQty / inciFac

		}

	} elseif (pTCodeArr.%Size() = 0) {
		s sQty = bQty
		s trdnFlag = 1
		if (uomType = "1") {
			s qty = sQty / inciFac
			s sp = bSp * inciFac

			q:(ingdRecCh = "") ""
		}
	 	s produceDate = $p($g(^DHCINGR(ingdRecId,"GRI",ingdRecCh)),"^",49)	 	
	}
	if (produceDate '= "") {
		s produceDate = $zd(produceDate,3)
	}
 	q produceDate
}

/// Description:通过业务指针获取已记录的追溯码  
/// Debug:      w ##class(PHA.FACE.TPS.INSUTE.Data).GetTraceCode("588117").%ToJSON()
ClassMethod GetTraceCode(Pointer)
{
	s retObj = ##class(PHA.COM.Object).%New()
	q:('$d(^DHCINTR(Pointer))&&(Pointer'["||")) retObj    
	if (Pointer["||") {
		s oeori = ""
	} else {
		s oeori = ..GetTransOrdItem(Pointer)
	}
	// 对于发退药类业务拆零按照配置，非发退药就按照实际是否拆零
	if (oeori '= "") {
		s inclb = $p(^DHCINTR(Pointer), "^", 7)
		s locId = $$LOC^ST01(inclb)
		s hospId = $p($g(^CTLOC(+locId)),"^",22)
		s paramStr = "^^^"_hospId
		s appName = ..%GetParameter("AppName")
		s tCodeWay = ##class(web.DHCST.Common.AppCommon).GetAppPropValue(appName, "GetTraceCodeWay", paramStr)
		if ((tCodeWay = "1") || (tCodeWay = "2") || (tCodeWay = "3")) {
			s ret = ..GetTraceCodeByAct(Pointer, tCodeWay)
		} elseif ((tCodeWay = "4") || (tCodeWay = "5")) {
			s ret = ..GetTraceCodeByUom(Pointer, tCodeWay)
		} else {
			s ret = ..GetTraceCodeByAct(Pointer, "1")
		}
	} else {
		s ret = ..GetTraceCodeByAct(Pointer, "1")
	}
	q ret
}

/// Description:通过实际业务判断拆零标识
/// Others:		配置1，非拆零单个码，拆零码个数和拆零数相等
/// 			配置2，非拆零和拆零码个数和业务数量相等
/// 			配置3，非拆零和拆零都传单个码 
/// Debug:      w ##class(PHA.FACE.TPS.INSUTE.Data).GetTraceCodeA("15263194")
ClassMethod GetTraceCodeByAct(Pointer, TCodeWay)
{
	s retObj = ##class(PHA.COM.Object).%New()
	q:('$d(^DHCINTR(Pointer))&&(Pointer'["||")) retObj    
	if (Pointer["||") {
		s type = "AT"
		s inclb = $p(^DHCINST(+Pointer, "STI", $p(Pointer, "||", 2)), "^", 1)
	} else {
		s type = $p(^DHCINTR(Pointer), "^", 1) 
		s inclb = $p(^DHCINTR(Pointer), "^", 7)
	}
	s locId = $$LOC^ST01(inclb)
	s hospId = $p($g(^CTLOC(+locId)),"^",22)
	s hospProName = $lg(..GetHospAddress(hospId), 2)
	s sTraceArr = ##class(PHA.COM.Array).%New()		// 拆零
	s pTraceArr = ##class(PHA.COM.Array).%New()		// 不拆零
	s (sQty, pQty) = 0
	s tbpId = ""
	for {
		s tbpId = $o(^INCTBPi("TypePointer",type,Pointer,tbpId))
		q:(tbpId = "")
		continue:('$d(^INCTBP(tbpId)))
		s codeObj = ##class(PHA.COM.Object).%New()
		s barCode = $p(^INCTBP(tbpId), "^", 9)
		d codeObj.%Set("drug_trac_codg", barCode)
		s bQty = $p(^INCTBP(tbpId), "^", 4)
		s trdnFlag = $p(^INCTBP(tbpId), "^", 12)
		if hospProName [ "江西"  {
			d codeObj.%Set("trdn_flag", trdnFlag)
		}
		if (trdnFlag = 1) {
			if (TCodeWay = "1") || (TCodeWay = "2") {
				for codeI = 1 : 1 : bQty {
					d sTraceArr.Push(codeObj)
				}
			} else {
				d sTraceArr.Push(codeObj)
			}
			s sQty = sQty + bQty
		} else {
			if (TCodeWay = "2") {
				for codeI = 1 : 1 : bQty {
					d pTraceArr.Push(codeObj)
				}
			} else {
				d pTraceArr.Push(codeObj)
			}
			s pQty = pQty + bQty
		}
	}
	/* 无码药品的处理-全归在拆零中 */
	s noTCodeFlag = "N"
	if ((pTraceArr.%Size() = 0) && (sTraceArr.%Size() = 0)) {
		s sQty = 0
		s inciId = +inclb
		s noTraceCodeFlag = ..GetIncitmPropValue(inciId, "NoTraceCode")
		if (noTraceCodeFlag = "Y") {
			s codeObj = ##class(PHA.COM.Object).%New()
			s paramStr = "^^^"_hospId
			s appName = ..%GetParameter("AppName")
			s barCode = ##class(web.DHCST.Common.AppCommon).GetAppPropValue(appName, "NoTraceCodeVal", paramStr)
			d codeObj.%Set("drug_trac_codg", barCode)
			if hospProName [ "江西"  {
				d codeObj.%Set("trdn_flag", trdnFlag)
			}
			s bUomId = $p(^INCI(inciId, 1), "^", 10)
			if (type = "AT") {
				s instQty = $p(^DHCINST(+Pointer, "STI", $p(Pointer, "||", 2)), "^", 2)
				s instUomId = $p(^DHCINST(+Pointer, "STI", $p(Pointer, "||", 2)), "^", 17)
				s instFac = ..UomFac(instUomId, bUomId)
				s bQty = instQty * instFac
			} else {
				s intrQty = $ZABS($p(^DHCINTR(Pointer), "^", 6))
				s intrUomId = $p(^DHCINTR(Pointer), "^", 10)
				s intrFac = ..UomFac(intrUomId, bUomId)
				s bQty = intrQty * intrFac
			}
			s noTCodeFlag = "Y"
			if (TCodeWay = "1") || (TCodeWay = "2") {
				for codeI = 1 : 1 : bQty {
					d sTraceArr.Push(codeObj)
				}
			} else {
				d sTraceArr.Push(codeObj)
			}
			s sQty = bQty
		}
	}
	d retObj.%Set("SQty", sQty)
	d retObj.%Set("PQty", pQty)
	d retObj.%Set("STCode", sTraceArr)
	d retObj.%Set("PTCode", pTraceArr)
	d retObj.%Set("NoTCodeFlag", noTCodeFlag)
	q retObj
}

/// Description:拆零标识通过基本单位和码类型包装单位是否一致来判断，一致非拆零标识为0，不一致拆零标识为1	
/// Others:		配置4，非拆零上传单个追溯码，拆零重复码个数和拆零数相等（陕西西安东软医保要求的规则）
/// 			配置5，非拆零和拆零都传单个码
/// Debug:      w ##class(PHA.FACE.TPS.INSUTE.Data).GetTraceCodeD("11811373")
ClassMethod GetTraceCodeByUom(Pointer, TCodeWay)
{
	s retObj = ##class(PHA.COM.Object).%New()
	q:('$d(^DHCINTR(Pointer))&&(Pointer'["||")) retObj    
	if (Pointer["||") {
		s type = "AT"
		s inclb = $p(^DHCINST(+Pointer, "STI", $p(Pointer, "||", 2)), "^", 1)
	} else {
		s type = $p(^DHCINTR(Pointer), "^", 1) 
		s inclb = $p(^DHCINTR(Pointer), "^", 7)
	}
	s locId = $$LOC^ST01(inclb)
	s hospId = $p($g(^CTLOC(+locId)),"^",22)
	s hospProName = $lg(..GetHospAddress(hospId), 2)
	s inciId = +inclb
	s bUomId = $p(^INCI(inciId,1), "^", 10)
	s pUomId = $p(^INCI(inciId,1), "^", 17)
	if (pUomId = "") {
		s pUomId = $p(^INCI(inciId,3), "^", 6)
	}
	if (pUomId = bUomId) {
		s trdnFlag = 0
	} else {
		s trdnFlag = 1
	}
	s sTraceArr = ##class(PHA.COM.Array).%New()		// 拆零
	s pTraceArr = ##class(PHA.COM.Array).%New()		// 不拆零
	s (sQty, pQty) = 0
	s tbpId = ""
	for {
		s tbpId = $o(^INCTBPi("TypePointer",type,Pointer,tbpId))
		q:(tbpId = "")
		continue:('$d(^INCTBP(tbpId)))
		s codeObj = ##class(PHA.COM.Object).%New()
		s barCode = $p(^INCTBP(tbpId), "^", 9)
		/* 针对非小码的码类型应重新判断 */
		s codeTypeDr = $p(^INCTBP(tbpId), "^", 11)
		s codeType = $p($g(^DHCSTSCDI(+codeTypeDr)), "^", 1)
		if (codeType '= "STC") {
			s codeUomId = pUomId
			s infoId = $o(^DHCITMINFO(0, "INCI", inciId, 0))
			if (codeType = "BTC") {
				s codeUomId = $p($g(^DHCITMINFO(+infoId)), "^", 51)
			} elseif (codeType = "MTC") {
				s codeUomId = $p($g(^DHCITMINFO(+infoId)), "^", 88)
			}
			if (codeUomId = bUomId) {
				s trdnFlag = 0
			} else {
				s trdnFlag = 1
			}
		}
		d codeObj.%Set("drug_trac_codg", barCode)
		if hospProName [ "江西"  {
			d codeObj.%Set("trdn_flag", trdnFlag)
		}
		s bQty = $p(^INCTBP(tbpId), "^", 4)
		if (trdnFlag = 1) {
			if (TCodeWay = "4") {
				for codeI = 1 : 1 : bQty {
					d sTraceArr.Push(codeObj)
				}
			} else {
				d sTraceArr.Push(codeObj)
			}
			s sQty = sQty + bQty
		} else {
			d pTraceArr.Push(codeObj)
			s pQty = pQty + bQty
		}
	}
	/* 无码药品的处理*/
	s noTCodeFlag = "N"
	if ((pTraceArr.%Size() = 0) && (sTraceArr.%Size() = 0)) {
		s (sQty, pQty) = 0
		s noTraceCodeFlag = ..GetIncitmPropValue(inciId, "NoTraceCode")
		if (noTraceCodeFlag = "Y") {
			s codeObj = ##class(PHA.COM.Object).%New()
			s paramStr = "^^^"_hospId
			s appName = ..%GetParameter("AppName")
			s barCode = ##class(web.DHCST.Common.AppCommon).GetAppPropValue(appName, "NoTraceCodeVal", paramStr)
			d codeObj.%Set("drug_trac_codg", barCode)
			if hospProName [ "江西"  {
				d codeObj.%Set("trdn_flag", trdnFlag)
			}
			if (type = "AT") {
				s instQty = $p(^DHCINST(+Pointer, "STI", $p(Pointer, "||", 2)), "^", 2)
				s instUomId = $p(^DHCINST(+Pointer, "STI", $p(Pointer, "||", 2)), "^", 17)
				s instFac = ..UomFac(instUomId, bUomId)
				s bQty = instQty * instFac
			} else {
				s intrQty = $ZABS($p(^DHCINTR(Pointer), "^", 6))
				s intrUomId = $p(^DHCINTR(Pointer), "^", 10)
				s intrFac = ..UomFac(intrUomId, bUomId)
				s bQty = intrQty * intrFac
			}
			s noTCodeFlag = "Y"
			if (trdnFlag = 1) {
				if (TCodeWay = "4") {
					for codeI = 1 : 1 : bQty {
						d sTraceArr.Push(codeObj)
					}
				} else {
					d sTraceArr.Push(codeObj)
				}
				s sQty = bQty
			} else {
				d pTraceArr.Push(codeObj)
				s pQty = bQty
			}
		}
	}
	d retObj.%Set("SQty", sQty)
	d retObj.%Set("PQty", pQty)
	d retObj.%Set("STCode", sTraceArr)
	d retObj.%Set("PTCode", pTraceArr)
	d retObj.%Set("NoTCodeFlag", noTCodeFlag)
	q retObj
}
