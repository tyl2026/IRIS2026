ClassMethod ybhljlExecute(ByRef qHandle As %Binary, SDate, EDate) As %Status
{

    Set qHandle=$lb(0,repid,0)
    Set ind=1

	s EDate=##class(websys.Conversions).DateHtmlToLogical(EDate)
	s EmrCodeStr="DHCNURYBHLJLDLR"

	s NLY=0
	s YLJGDM=$p(##Class(web.DHCENS.BLL.WSJ.Tool.AdmInfo).Hospital(),"^",1)
	f EmrDate=SDate:1:EDate d
	.s Len=$l(EmrCodeStr,"^")
	.f i=1:1:Len  d	 
	..s EmrCode=$p(EmrCodeStr,"^",i)
	..s EmrCode=$ZConvert(EmrCode,"U")
	..s TemplateId=""
	..s TemplateId=$o(^NurMp.TemplateI("Indentity"," "_EmrCode,TemplateId))
	..q:TemplateId="" 
	..s a=##class(NurMp.Template).%OpenId(TemplateId) 
	..s TCIndent=a.TCIndentity
	..s Typ=$ZConvert(TCIndent,"U")
	..s PGDID=0  f  s PGDID=$o(^NurMp.DHCTempMultDataI("emdate",EmrDate," "_Typ,PGDID)) q:PGDID=""  d
	...s arr=##class(NurMp.DHCTempMultData).%OpenId(PGDID)
    ...s RecUser=arr.RecCancelUser //作废
    ...q:RecUser'=""
    ...s adm=arr.EpisodeId
    ...s PatRowID=$p($g(^PAADM(adm)),"^",1)
    ...s PAADMType=$p($g(^PAADM(adm)),"^",2)
    ...q:PAADMType'="I"
	...s patInfo=##Class(web.DHCENS.BLL.WSJ.Tool.AdmInfo).Patinfo(PatRowID)
	...s PatNo=$p(^PAPER(PatRowID,"PAT",1),"^",2) ;登记号
	...s JZLSH= adm // 就诊流水号
	...s JLH=PGDID  //记录号
	...s HZSFZJHM=$p($g(^PAPER(PatRowID,"PAT",3)),"^",6)  //证件号码 必填 身份证号
	...I HZSFZJHM="" S HZSFZJHM=PatNo
	...s SFZJLBDM="01"
	...s:$l(HZSFZJHM)'=18 SFZJLBDM="99"
	...s KH=$p(patInfo,"^",7)
	...i KH="" s KH=PatNo  //卡号
	...s KLX=$p(patInfo,"^",6)
	...s HZLX="1"
	...i KLX="" s KLX=6   //卡类型
	...s ZYH=$p(patInfo,"^",3) //住院号 必填
	...s BAH=$p(patInfo,"^",3) //病案号 必填
	...s PatCardType=$p($g(^PAPER(PatRowID,"PAT",3)),"^",7)
	...i PatCardType'="" d
	....if PatCardType="21" s ZJLX="01"
	....else  s ZJLX=$p($g(^PAC("CARD",PatCardType)),"^",1)
	...s XB=$p(patInfo,"^",4)
	...s HZXM=$p(patInfo,"^",1)
	...s HYZK=$p($g(^PAPER(PatRowID,"PER",2)),"^",3) 
	...i HYZK'="" d
	....s HYData=$g(^CT("MAR",HYZK))
	....s HYZK=$p($g(HYData),"^",1)
	....s HYZKMC=$p($g(HYData),"^",2)
	....s:HYZK="" HYZK="90",HYZKMC="未说明的婚姻状况"
	...s NLSS= $p(patInfo,"^",5) //必填 年龄
	...s admdepdr=$p($g(^PAADM(adm)),"^",4)
	...s KSBM=$p(^CTLOC(admdepdr),"^",1) //必填 入院科室代码
	...s KSMC= $p(^CTLOC(admdepdr),"^",2) //必填 入院科室名称
	...s HLDJDM=##class(Nur.NIS.Service.Base.Patient).GetCareLevel(adm)
	...s:HLDJDM="" HLDJDM="一级护理"
	...s RecWard=arr.RecWard
	...s:RecWard'="" BQMC=$p(^PAWARD(RecWard),"^",2)
	...s BCH=##class(Nur.NIS.Service.Base.Patient).GetBedCode(adm)
	...s JZSJ= $ZD($P(^PAADM(adm),"^",6),3) //必填 诊断时间
	...s cbdiagstr=##class(web.DHCENS.BLL.WSJ.Tool.AdmInfo).GetDiagnoses(adm)
	...s JBZDDM= $P(cbdiagstr,"^",4) //必填 疾病诊断代码（初步诊断)
	...s JBZDMC= $P(cbdiagstr,"^",5) //必填 疾病诊断名称（初步诊断)
	...s TZ=##class(web.DHCDoc.OP.AjaxInterface).GetPatWeight(adm)
    ...s TW=$ZOBJPROPERTY(arr,"Item4")   //体温
    ...s HXPL=$ZOBJPROPERTY(arr,"Item6")   //呼吸
    ...s ML=$ZOBJPROPERTY(arr,"Item5")   //脉搏
    ...s SSY=$ZOBJPROPERTY(arr,"Item7")   //收缩
    ...s SZY=$ZOBJPROPERTY(arr,"Item26")   //舒张
    ...s XYBHD=$ZOBJPROPERTY(arr,"Item8")   //血氧饱和度
    ...s ZBDMBDBZ=""
    ...s YSQKDM=""
    ...s GD=$ZOBJPROPERTY(arr,"Item18")		//管道
    ...s DGHLMS=##class(web.LFTest).formatValueNew(GD)
    ...s QGHLDM=""
    ...s TWHL=$ZOBJPROPERTY(arr,"Item30")   //体位
    ...s PF=$ZOBJPROPERTY(arr,"Item21")		//皮肤
    ...s PFHL=##class(web.LFTest).formatValueNew(PF)
    ...s YYHL=""
    ...s YSZDDM=""
    ...s XLHLDM=""
    ...s AQHLDM=""
    ...s JYBQ=$ZOBJPROPERTY(arr,"Item3")  //病情
    ...s HLGCXMMC=""
    ...s HLGCJG=""

    ...s QMRQSJ=$zd(EmrDate,3)_" "_$zt(EmrTime)

    ...s SFDD=""

    ...s DRUGFALL=""

ClassMethod ybhljlClose(ByRef qHandle As %Binary) As %Status [ PlaceAfter = ybhljlExecute ]
{
	Set repid=$LIST(qHandle,2)
	Kill ^CacheTemp(repid)
	Quit $$$OK
}

ClassMethod ybhljlFetch(ByRef qHandle As %Binary, ByRef Row As %List, ByRef AtEnd As %Integer = 0) As %Status [ PlaceAfter = ybhljlExecute ]
{
	Set AtEnd=$LIST(qHandle,1)
	Set repid=$LIST(qHandle,2)
	Set ind=$LIST(qHandle,3)
	Set ind=$o(^CacheTemp(repid,ind))
	If ind="" {				
		Set AtEnd=1
		Set Row=""
	}
	Else      {				
		Set Row=^CacheTemp(repid,ind)
	}
	s qHandle=$lb(AtEnd,repid,ind)
	;b
	Quit $$$OK
}

ClassMethod pgcssjlClose(ByRef qHandle As %Binary) As %Status [ PlaceAfter = pgcssjlExecute ]
{
	Set repid=$LIST(qHandle,2)
	Kill ^CacheTemp(repid)
	Quit $$$OK
}

ClassMethod pgcssjlFetch(ByRef qHandle As %Binary, ByRef Row As %List, ByRef AtEnd As %Integer = 0) As %Status [ PlaceAfter = pgcssjlExecute ]
{
	Set AtEnd=$LIST(qHandle,1)
	Set repid=$LIST(qHandle,2)
	Set ind=$LIST(qHandle,3)
	Set ind=$o(^CacheTemp(repid,ind))
	If ind="" {				
		Set AtEnd=1
		Set Row=""
	}
	Else      {				
		Set Row=^CacheTemp(repid,ind)
	}
	s qHandle=$lb(AtEnd,repid,ind)
	;b
	Quit $$$OK
}
