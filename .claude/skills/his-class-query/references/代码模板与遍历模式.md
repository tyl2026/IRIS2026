# Class Query 代码模板与遍历模式

## 代码模板

### 标准 Query 模板（Execute/Fetch/Close）

```objectscript
/// 查询说明
/// d ##class(%ResultSet).RunQuery("web.ClassName","QueryName","参数1","参数2")
Query QueryName(Param1 As %String, Param2 As %String) As %Query(ROWSPEC = "Field1,Field2:%Float") [ SqlProc ]
{
}

ClassMethod QueryNameExecute(ByRef qHandle As %Binary, Param1 As %String, Param2 As %String) As %Status
{
    Set repid=$I(^CacheTemp)
    Set qHandle=$lb(0,repid,0)
    Set ind=1

    s Param1=$zdh(Param1,3)
    s Param2=$zdh(Param2,3)
    q:(Param1="")||(Param2="") $$$OK

    ; 变量初始化
    s (Field1,Field2)=""

    f Date=Param1:1:Param2 d
    .s RowId="" f  s RowId=$o(^GlobalNode(Date,RowId)) q:RowId=""  d
    ..s NodeData=$g(^GlobalNode(RowId))
    ..q:NodeData=""
    ..s Field1=$p(NodeData,"^",1)  //字段1
    ..s Field2=$p(NodeData,"^",2)  //字段2
    ..q:Field1=""
    ..d OutputRow

    Quit $$$OK

OutputRow
    s Data=$lb(Field1,Field2)
    s ^CacheTemp(repid,ind)=Data
    s ind=ind+1
    q
}

ClassMethod QueryNameFetch(ByRef qHandle As %Binary, ByRef Row As %List, ByRef AtEnd As %Integer = 0) As %Status [ PlaceAfter = QueryNameExecute ]
{
    Set AtEnd=$LIST(qHandle,1)
    Set repid=$LIST(qHandle,2)
    Set ind=$LIST(qHandle,3)
    Set ind=$o(^CacheTemp(repid,ind))
    If ind="" {
        Set AtEnd=1
        Set Row=""
    }
    Else {
        Set Row=^CacheTemp(repid,ind)
    }
    s qHandle=$lb(AtEnd,repid,ind)
    Quit $$$OK
}

ClassMethod QueryNameClose(ByRef qHandle As %Binary) As %Status [ PlaceAfter = QueryNameExecute ]
{
    Set repid=$LIST(qHandle,2)
    Kill ^CacheTemp(repid)
    Quit $$$OK
}
```

### 汇总统计模板（使用临时 Global 去重/累加）

```objectscript
ClassMethod QueryNameExecute(ByRef qHandle As %Binary, ...) As %Status
{
    Set repid=$I(^CacheTemp)
    Set qHandle=$lb(0,repid,0)
    Set ind=1
    k ^TEMP($j)  ; 清理临时 Global

    ; 第一遍：遍历数据，按维度累加到临时 Global
    f Date=startDate:1:endDate d
    .s RowId="" f  s RowId=$o(^Global(Date,RowId)) q:RowId=""  d
    ..s NodeData=$g(^Global(RowId))
    ..q:NodeData=""
    ..s DimKey=$p(NodeData,"^",1)
    ..s ^TEMP($j,DimKey)=$g(^TEMP($j,DimKey))+1

    ; 第二遍：从临时 Global 输出结果
    s DimKey="" f  s DimKey=$o(^TEMP($j,DimKey)) q:DimKey=""  d
    .s Count=$g(^TEMP($j,DimKey))
    .d OutputRow

    k ^TEMP($j)  ; 清理
    q $$$OK
OutputRow
    s Data=$lb(DimKey,Count)
    s ^CacheTemp(repid,ind)=Data
    s ind=ind+1
    q
}
```

### ClassMethod 直接输出模板（非 Query，用于简单场景）

```objectscript
/// w ##class(web.ClassName).MethodName("参数")
ClassMethod MethodName(Param1 As %String) As %String
{
    q:Param1="" ""
    s Result=""
    s RowId="" f  s RowId=$o(^Global(Param1,RowId)) q:RowId=""  d
    .s NodeData=$g(^Global(RowId))
    .q:NodeData=""
    .s Field1=$p(NodeData,"^",1)
    .s Field2=$p(NodeData,"^",2)
    .w !,Field1_","_Field2
    q ""
}
```

## 核心 Global 遍历模式

### 按日期索引遍历

```objectscript
; ^GlobalNamei("DateIndex", Date, RowId)
f Date=startDate:1:endDate d
.s RowId="" f  s RowId=$o(^GlobalNamei("DateIndex",Date,RowId)) q:RowId=""  d
```

### 按就诊遍历医嘱

```objectscript
; ^OEORD(0,"Adm", AdmId, OrderRowId)
s OrderRowId=$o(^OEORD(0,"Adm",AdmId,""))
s Sub=0 f  s Sub=$o(^OEORD(OrderRowId,"I",Sub)) q:Sub=""  d
```

### 按账单遍历明细

```objectscript
; ^DHCPB(BillId,"O",OrdSub) — 医嘱子表
; ^DHCPB(BillId,"O",OrdSub,"D",DetSub) — 明细子表
f  s OrdSub=$o(^DHCPB(BillId,"O",OrdSub)) q:OrdSub=""  d
.f  s DetSub=$o(^DHCPB(BillId,"O",OrdSub,"D",DetSub)) q:DetSub=""  d
```

### 按病人遍历就诊

```objectscript
; ^PAPERdr(PAPMIDR,"ADM", AdmType, AdmRowId)
s AdmType="" f  s AdmType=$o(^PAPERdr(PAPMIDR,"ADM",AdmType)) q:AdmType=""  d
.s AdmRowId="" f  s AdmRowId=$o(^PAPERdr(PAPMIDR,"ADM",AdmType,AdmRowId)) q:AdmRowId=""  d
```

### 子表遍历（病区→床位）

```objectscript
; ^PAWARD(WardId,"BED",BedSub)
f  s WardId=$o(^PAWARD(WardId)) q:+WardId=0  d
.s BedSub="" f  s BedSub=$o(^PAWARD(WardId,"BED",BedSub)) q:BedSub=""  d
```

### 在院住院遍历

```objectscript
; ^CTLOC(0,"DepGrp",2,LocId) — 科室组（临床科室=2）
; ^PAADMi("NotCompl","I",LocId,AdmId) — 未完成的住院就诊索引
f  s LocId=$o(^CTLOC(0,"DepGrp",2,LocId)) q:LocId=""  d
.s AdmId="" f  s AdmId=$o(^PAADMi("NotCompl","I",LocId,AdmId)) q:AdmId=""  d
```

### 诊断子表遍历

```objectscript
; ^MRi("Date",Date,MRRowId) — 按日期索引遍历病历
; ^MR(MRRowId,"DIA",DiaSub) — 诊断子表（^1=ICD诊断DR）
; ^MRC("ID",ICDDR) — ICD诊断字典（^2=描述）
f Date=startDate:1:endDate d
.s MRRowId="" f  s MRRowId=$o(^MRi("Date",Date,MRRowId)) q:MRRowId=""  d
..s DiaSub="" f  s DiaSub=$o(^MR(MRRowId,"DIA",DiaSub)) q:DiaSub=""  d
...s ICDDR=$p($g(^MR(MRRowId,"DIA",DiaSub)),"^",1)
...s DiagDesc=$p($g(^MRC("ID",ICDDR)),"^",2)
```

### 医嘱执行记录遍历

```objectscript
; ^OEORDi(0,"StDt",Date,OERowId,OEORIsub) — 按开始日期查医嘱

...s DSPRowId="" f  s DSPRowId=$o(^DHCOEDISQTY(0,"OEORI",OEORIRowId,DSPRowId)) q:DSPRowId=""  d
```

## 常用取值模式

| 需求 | 代码 |
|------|------|
| 取就诊信息 | `s PAADMData=$g(^PAADM(AdmId))` → `s AdmType=$p(PAADMData,"^",2)`, `s VisitStatus=$p(PAADMData,"^",20)` |
| 取病人姓名 | `s PAPMIDR=$p(PAADMData,"^",1)` → `s Name=$p($g(^PAPER(PAPMIDR,"ALL")),"^",1)` |
| 取科室名称 | `s LocDR=$p(PAADMData,"^",4)` → `s LocDesc=$p($g(^CTLOC(LocDR)),"^",2)` |
| 取医生姓名 | `s DocDR=$p(PAADMData,"^",9)` → `s DocName=$p($g(^CTPCP(DocDR,1)),"^",2)` |
| 取年龄 | `s Age=##class(EMRservice.HISInterface.PatientInfoAssist).Age(PapmiDR,AdmID,"","")` |
| 取医嘱名称 | `s ARCIMSub=$p(ItemMastDR,"\|\|",2)` → `s Desc=$p($g(^ARCIM(+ItemMastDR,ARCIMSub,1)),"^",2)` |
| 取就诊状态 | `s Status=$p(PAADMData,"^",20)` （A=在院, C=取消, D=出院） |
| 取就诊类型 | `s Type=$p(PAADMData,"^",2)` （O=门诊, E=急诊, I=住院） |
| 日期格式化 | `s DateStr=$zd(Date,3)` → `"2024-01-01"` |
| 时间格式化 | `s TimeStr=$zt(Time,2)` → `"14:30:00"` |
| 取诊断描述 | `s ICDDR=$p($g(^MR(Adm,"DIA",sub)),"^",1)` → `s DiagDesc=$p($g(^MRC("ID",ICDDR)),"^",2)` |
| 取收费项目 | `s TarName=$p($g(^DHCTARI(TarItemDR)),"^",2)` |
| 取床位类型 | `s TypeName=$p($g(^PAC("BEDTP",TypeDR)),"^",2)` |
| 取民族 | `s NationName=$p($g(^CT("NAT",NationDR)),"^",2)` |
| 取用户姓名 | `s UserName=$p($g(^SSU("SSUSR",UserDR)),"^",2)` |
