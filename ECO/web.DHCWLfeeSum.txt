/// 根据医嘱id获取执行天数
/// w ##class(web.DHCWLfeeSum).getZXDAYS("7629679||532")
ClassMethod getZXDAYS(oeori)
{
	k ^||TEMPDHCWL($j,"getZXDAYS")
	//q 0
	s zxdays=0
	q:oeori="" 0
	s ord=$p(oeori,"||",1)
	s tsub=$p(oeori,"||",2)
	s arcid=$p(^OEORD(ord,"I",tsub,1),"^",2)
	q:arcid="" 0
	s exsdate=""
	f  s exsdate=$O(^OEORDi(0,"ARCIM",ord,arcid,exsdate))  q:exsdate=""  d
	.q:+exsdate=0
	.s sub=""
	.f  s sub=$O(^OEORDi(0,"ARCIM",ord,arcid,exsdate,sub))  q:sub=""  d
	..q:+sub=0
	..q:'$d(^OEORD(ord,"I",sub,"X"))
	..s ore=""
	..f  s ore=$o(^OEORD(ord,"I",sub,"X",ore)) q:ore=""  d
	...q:+ore=0
	...s execdata=$g(^OEORD(ord,"I",sub,"X",ore))
	...s OEOREStatus=$p(execdata,"^",16)
	...q:OEOREStatus'="1"
	...s exsdate2=$p(execdata,"^",1)
	...i '$d(^||TEMPDHCWL($j,"getZXDAYS",exsdate2)) d
	....s zxdays=+zxdays+1
	....s ^||TEMPDHCWL($j,"getZXDAYS",exsdate2)=""
	q zxdays
}

/// 根据医嘱id获取执行天数
/// w ##class(web.DHCWLfeeSum).getZXDAYS2("7139279||91||1")
ClassMethod getZXDAYS2(oeore)
{
	
	s zxdays="",dayct=0
	//q:oeore="" 0
	s ord=$p(oeore,"||",1)
	s tsub=$p(oeore,"||",2)
	s ore=$p(oeore,"||",3)
	s exsdate=$p($g(^OEORD(ord,"I",tsub,"X",ore)),"^",1)
	s freqdr=$p(^OEORD(ord,"I",tsub,2),"^",4)
	s freq=$p($g(^PHCFR(freqdr)),"^",1)
	s factor=+$p($g(^PHCFR(freqdr)),"^",2)
	i factor=0 s factor=1
	s arcid=$p(^OEORD(ord,"I",tsub,1),"^",2)
	
	//^OEORDi(0,"DateARCIM",{OEORE_ExStDate},$P($G(^OEORD({OE_Order.OEORD_RowId},"I",{OE_OrdItem.OEORI_Childsub},1)),"^",2),{OE_Order.OEORD_RowId},{OE_OrdItem.OEORI_Childsub},{OEORE_Childsub})
	
	q:arcid="" 0
	s sub=""
	f  s sub=$O(^OEORDi(0,"DateARCIM",exsdate,arcid,ord,sub))  q:sub=""  d
	.s ore=""
	.f  s ore=$O(^OEORDi(0,"DateARCIM",exsdate,arcid,ord,sub,ore)) q:ore=""  d
	..q:+ore=0
	..s execdata=$g(^OEORD(ord,"I",sub,"X",ore))
	..s OEOREStatus=$p(execdata,"^",16)
	..q:OEOREStatus'="1"
	..s dayct=dayct+1
	s ret=freq_" "_$zd(exsdate,3)_"："_dayct_"次"
	i factor<dayct s ret=ret_"(异常)"
	q ret
}

/// 根据收费项id获取单日数量
/// w ##class(web.DHCWLfeeSum).gettardayct(7186178,5915,67255)
/// w ##class(web.DHCWLfeeSum).gettardayct(7186201,5915,67255)
ClassMethod gettardayct(adm2 = "", tarid2 = "", date2 = "")
{
	s dayct2=0,admct2=0
	q:(adm2 = "")||(tarid2 = "")||(date2 = "") 0
	q:'$d(^DHCWorkLoad(0,"PAADM",adm2)) 0
	s wlid2=""
	f  s wlid2=$o(^DHCWorkLoad(0,"PAADM",adm2,wlid2)) q:wlid2=""  d
	.q:+wlid2=0
	.s qty2=+$P($g(^DHCWorkLoad(wlid2)),"^",15)
	.s tardr2=$P(^DHCWorkLoad(wlid2),"^",22)
	.q:tardr2'=tarid2
	.s admct2=admct2+qty2
	.s ordate2=$P(^DHCWorkLoad(wlid2),"^",5)
	.q:(date2 '= "")&&(ordate2 '= date2)
	.s dayct2=dayct2+qty2
	q dayct2_"(共计:"_admct2_")"
}

    ..q:$D(^||TEMPDHCWL($j,"Arcitems",ItemOrdr))
    ..s WLRowid=""
    ..f  s WLRowid=$O(^DHCWorkLoad(0,"DateItemOrd",day2,ItemOrdr,WLRowid)) q:WLRowid=""  d

    ...s qty=$P(^DHCWorkLoad(WLRowid),"^",15)
    ...s type=$P(^DHCWorkLoad(WLRowid),"^",4)
    ...s adm=$P(^DHCWorkLoad(WLRowid),"^",12)
    ...s oeore=$P(^DHCWorkLoad(WLRowid),"^",54)
    ...s oeori=$P(^DHCWorkLoad(WLRowid),"^",21)
    ...s arcdesc=$p(^ARCIM(+ItemOrdr,$p(ItemOrdr,"||",2),1),"^",2)
    ...s zflag=0
    ...i oeore'="" s zflag=..feeflag(oeore)

    ....s ^||TEMPDHCWL($j,"Arcitems2-day-desc",adm,day2)=$g(^||TEMPDHCWL($j,"Arcitems2-day-desc",adm,day2))_","_arcdesc

	...s arcdesc=$P(^ARCIM(+arcid,$p(arcid,"||",2),1),"^",2)

ClassMethod getfeesumFetch(ByRef qHandle As %Binary, ByRef Row As %List, ByRef AtEnd As %Integer = 0) As %Status [ PlaceAfter = getfeesumExecute ]
{
 	s AtEnd=$li(qHandle,1)
 	s repid=$li(qHandle,2)
 	s ind=$li(qHandle,3)
 	s ind=$o(^CacheTemp(repid,ind))
 	if ind="" {				// if there are no more rows, finish fetching
 		s AtEnd=1
 		s Row=""
 	}
 	else      {				// fetch row
 		s Row=^CacheTemp(repid,ind)
 	}
 	// Save QHandle
 	s qHandle=$lb(AtEnd,repid,ind)
	q $$$OK
}

ClassMethod getfeesumClose(ByRef qHandle As %Binary) As %Status [ PlaceAfter = getfeesumExecute ]
{
	s repid=$li(qHandle,2)
 	k ^CacheTemp(repid)
 	q $$$OK
}

/// 处理单条 WorkLoad 记录 — 公共筛选与数据提取
/// 供 getfeesumExecute 各分支调用，统一应用所有筛选条件
/// 返回: 1=已处理(需 count+1) / ""=被过滤
ClassMethod ProcessWLRecord(WLRowid As %String, day As %String, HZFlag As %String, ExtType As %String, drugtype As %String, ByRef count As %Integer) As %String [ Private ]
{
	s ret=""

	; --- 基础数据提取 ---
	s fee=$P(^DHCWorkLoad(WLRowid),"^",16)
	s price=$P(^DHCWorkLoad(WLRowid),"^",14)
	s qty=$P(^DHCWorkLoad(WLRowid),"^",15)
	s type=$P(^DHCWorkLoad(WLRowid),"^",4)

	; 账单筛选
	s pbddr=$P(^DHCWorkLoad(WLRowid),"^",48)
	s pbdr=+pbddr
	q:pbdr=0 ""
	s payflag=$p($g(^DHCPB(pbdr)),"^",16)
	s refundflag=$p($g(^DHCPB(pbdr)),"^",17)
	q:(payflag '= "P")&&(payflag '= "B") ""
	q:refundflag'="" ""

	s billdatetime=""
	i pbddr'="" d
	. s billdate=$p($g(^DHCPB(+pbddr,"O",$p(pbddr,"||",2),"D",$p(pbddr,"||",3))),"^",11)
	. s billtime=$p($g(^DHCPB(+pbddr,"O",$p(pbddr,"||",2),"D",$p(pbddr,"||",3))),"^",12)
	. s billdatetime=$zd(billdate,3)_" "_$zt(billtime,1)

	; 就诊类型筛选
	q:$d(^||TEMPDHCWL($j,"Atype"))&&'$d(^||TEMPDHCWL($j,"Atype",type)) ""

	; 患者信息
	s adm=$P(^DHCWorkLoad(WLRowid),"^",12)
	s papmidr=$P(^DHCWorkLoad(WLRowid),"^",13)
	s REGNO=$P(^PAPER(papmidr,"PAT",1),"^",2)
	s medno=$P(^PAPER(papmidr,"PAT",1),"^",22)
	s name=$P(^PAPER(papmidr,"ALL"),"^",1)
	s XB=""
	s sexId=$p($g(^PAPER(papmidr,"ALL")),"^",7)
	s:sexId'="" XB=$p(^CT("SEX",sexId),"^",2)
	s ZJH=$p($g(^PAPER(papmidr,"PAT",3)),"^",6)

	; 医嘱子分类筛选
	s TarCate=$p(^DHCWorkLoad(WLRowid),"^",9)
	q:$D(^||TEMPDHCWL($j,"ItemSubs"))&&'$D(^||TEMPDHCWL($j,"ItemSubs",TarCate)) ""

	; 费别检查
	s oeori=$p($g(^DHCWorkLoad(WLRowid)),"^",21)
	s oeore=$p($g(^DHCWorkLoad(WLRowid)),"^",54)
	s oeflag=..feeflag2(oeori)
	q:oeflag'=1 ""

	s pbdr=$p($g(^DHCWorkLoad(WLRowid)),"^",20)

	; 扩展医嘱项/收费项筛选
	s exqty=$g(^||TEMPDHCWL($j,"Arcitems2-adm",adm))
	s exqty2=$g(^||TEMPDHCWL($j,"Arcitems2-day",adm,day))
	q:(ExtType = "1")&&('(exqty > 0))&&($D(^||TEMPDHCWL($j,"Arcitems2"))) ""
	q:(ExtType = "2")&&('(exqty2 > 0))&&($D(^||TEMPDHCWL($j,"Arcitems2"))) ""

	s tarexqty=$g(^||TEMPDHCWL($j,"Taritems2-adm",adm))
	s tarexqty2=$g(^||TEMPDHCWL($j,"Taritems2-day",adm,day))
	q:(ExtType = "1")&&('(tarexqty > 0))&&($D(^||TEMPDHCWL($j,"Taritems2"))) ""
	q:(ExtType = "2")&&('(tarexqty2 > 0))&&($D(^||TEMPDHCWL($j,"Taritems2"))) ""

	; 住院天数
	s zyts=""
	s PAADMOutDate=$p($g(^PAADM(adm)),"^",59)
	s PAADMAdmDate=$p($g(^PAADM(adm)),"^",6)
	i PAADMOutDate'="" d
	. s zyts=PAADMOutDate-PAADMAdmDate
	e  d
	. s zyts=(+$h)-PAADMAdmDate
	i zyts="0" s zyts=1

	; 费别
	s admreason=""
	s admreasondr=$p(^PAADM(adm,1),"^",7)
	i admreasondr'="" s admreason=$p(^PAC("ADMREA",admreasondr),"^",2)

	; 医嘱日期
	s oesdate=$p(^OEORD(+oeori,"I",$p(oeori,"||",2),1),"^",9)
	s oestime=$p(^OEORD(+oeori,"I",$p(oeori,"||",2),1),"^",10)
	s oedatetime=$zd(oesdate,3)_" "_$zt(oestime,1)

	; 医保信息
	i (HZFlag = "1")||(HZFlag = "3") s insustr=""
	e  s insustr=..GetINSUInfo2(pbdr)
	s Iadmtype=$p(insustr,"^",1)
	s xzlx=$p(insustr,"^",2)
	s ydflag=$p(insustr,"^",3)

	; 医嘱项/收费项筛选
	s ItemOrdr=$P(^DHCWorkLoad(WLRowid),"^",2)
	q:$D(^||TEMPDHCWL($j,"Arcitems"))&&'$D(^||TEMPDHCWL($j,"Arcitems",ItemOrdr)) ""
	s tardr=$P(^DHCWorkLoad(WLRowid),"^",22)
	q:$D(^||TEMPDHCWL($j,"Taritems"))&&'$D(^||TEMPDHCWL($j,"Taritems",tardr)) ""

	; 药品分类筛选
	s incid=$o(^INCI(0,"ARCIM_DR",+ItemOrdr,""))
	q:(..getdrugtypebyinci(incid, drugtype)'=1)&&(drugtype '= "")&&(incid '= "") ""

	; 开单科室
	s patdepdr=$P(^DHCWorkLoad(WLRowid),"^",23)
	q:$D(^||TEMPDHCWL($j,"PatDeps"))&&'$D(^||TEMPDHCWL($j,"PatDeps",patdepdr)) ""
	s patdepdesc=$p(^CTLOC(patdepdr),"^",2)
	i patdepdesc["-" s patdepdesc=$p(patdepdesc,"-",2)

	; 接收科室
	s recdepdr=$P(^DHCWorkLoad(WLRowid),"^",1)
	q:$D(^||TEMPDHCWL($j,"RecDeps"))&&'$D(^||TEMPDHCWL($j,"RecDeps",recdepdr)) ""
	s recdepdesc=$p(^CTLOC(recdepdr),"^",2)
	i recdepdesc["-" s recdepdesc=$p(recdepdesc,"-",2)

	; 病人科室
	s resdepdr=$P(^DHCWorkLoad(WLRowid),"^",3)
	q:$D(^||TEMPDHCWL($j,"ResDeps"))&&'$D(^||TEMPDHCWL($j,"ResDeps",resdepdr)) ""
	s resdepdesc=$p(^CTLOC(resdepdr),"^",2)
	i resdepdesc["-" s resdepdesc=$p(resdepdesc,"-",2)

	; 收费子分类筛选
	s TARECDr=$p(^DHCWorkLoad(WLRowid),"^",41)
	q:$D(^||TEMPDHCWL($j,"ItemTarecs"))&&'$D(^||TEMPDHCWL($j,"ItemTarecs",TARECDr)) ""

	; 医生信息
	s ResDocDr=$P(^DHCWorkLoad(WLRowid),"^",19)
	s ResDoc=""
	i ResDocDr'="" S ResDoc=$p($g(^CTPCP(ResDocDr,1)),"^",2)
	i +$h>"67750" s ResDoc=""

	; --- 累加汇总 ---
	s ^||TEMPDHCWL($j,"QTYSUM",pbdr,oeori)=+$g(^||TEMPDHCWL($j,"QTYSUM",pbdr,oeori))+qty
	s ^||TEMPDHCWL($j,"FEESUM",ItemOrdr,tardr,price)=+$g(^||TEMPDHCWL($j,"FEESUM",ItemOrdr,tardr,price))+fee
	s ^||TEMPDHCWL($j,"FEESUM",ItemOrdr,tardr,price,"QTY")=+$g(^||TEMPDHCWL($j,"FEESUM",ItemOrdr,tardr,price,"QTY"))+qty

	; 非汇总模式：构建明细记录
	i HZFlag'="1" d
	. s tarcode=$P(^DHCTARI(tardr),"^",1)
	. s tardesc=$P(^DHCTARI(tardr),"^",2)
	. s arccode=$P(^ARCIM(+ItemOrdr,$p(ItemOrdr,"||",2),1),"^",1)
	. s arcdesc=$P(^ARCIM(+ItemOrdr,$p(ItemOrdr,"||",2),1),"^",2)
	. s partDesc=##class(web.DHCAPPInterface).GetExaReqPartDesc(oeori)
	. i partDesc'="" s arcdesc=arcdesc_"-"_partDesc
	. s arcstr=""
	. i (ExtType = "1")&&($D(^||TEMPDHCWL($j,"Arcitems2"))) s arcstr=$g(^||TEMPDHCWL($j,"Arcitems2-adm-desc",adm))
	. i (ExtType = "2")&&($D(^||TEMPDHCWL($j,"Arcitems2"))) s arcstr=$g(^||TEMPDHCWL($j,"Arcitems2-day-desc",adm,day))
	. s arcstr=arcdesc_arcstr
	. S admtype="住院"
	. i type="O"  S admtype="门诊"
	. i type="E"  S admtype="急诊"
	. i type="H"  S admtype="体检"
	. i (HZFlag = "3") s dayct=""
	. e  s dayct=..gettardayct(adm,tardr,day)
	. s ^||TEMPDHCWL($j,"Detail",count)=$lb(arccode,arcdesc,tarcode,tardesc,qty,price,fee,name,REGNO,patdepdesc,recdepdesc,ResDoc,admreason,billdatetime,oedatetime,Iadmtype,xzlx,ydflag,resdepdesc,oeori,pbdr,medno,admtype,adm,XB,ZJH,dayct,exqty2,zyts,arcstr,$zd(PAADMAdmDate,3),oeore)
	. s count=count+1

	q 1
}

/// 判断统计类型，入参类型代码、incid
/// w ##Class("web.HNYZDHCTYL").dhcwlflag("HuNanLcyyzk","tqmxhtjhs","4245")
ClassMethod getdrugtypebyinci(incid As %String, drugtypestr) As %String
{
	s flag=0
	q:incid="" flag
	q:'$d(^CD.PHA.IN.STAT("INCI",incid)) flag
	s typeid=""
    f  s typeid=$o(^CD.PHA.IN.STAT("INCI",incid,typeid)) q:typeid=""  d
    .s catid=""
    .f  s catid=$o(^CD.PHA.IN.STAT("INCI",incid,typeid,catid)) q:catid=""  d
    .q:(("^"_drugtypestr_"^")'[("^"_typeid_"||"_catid_"^"))
    .s flag=1
	q flag
}

/// ^DHCWorkLoad(0,"OEORE",{WorkLoad_OEORE_Dr},{WorkLoad_Rowid})
/// w ##class(web.DHCWLfeeSum).feeflag2("6850929||9")     
ClassMethod feeflag(orore As %String)
{
	q:orore="" "" 
	s ret ="",sqty=0
	q:'$d(^DHCWorkLoad(0,"OEORE",orore)) ""
	s wlid=""
	f  s wlid=$o(^DHCWorkLoad(0,"OEORE",orore,wlid)) q:wlid=""  d
	.q:+wlid=0
	.s qty=$P(^DHCWorkLoad(wlid),"^",15)
	.s sqty=sqty+qty
	i sqty>0 s ret=1
	q ret
}

	.q:+wlid=0
	.s qty=$P(^DHCWorkLoad(wlid),"^",15)
	.s sqty=sqty+qty
	i sqty>0 s ret=1
	q ret
}

/// 根据医嘱id获取当日数量
/// w ##class(web.HNYZDHCTYL8).getZXDAYS("7705595||19")
ClassMethod getdayct(oeori, date)
{
	
		
	s dayct=0
	q:oeori="" 0
	s ord=$p(oeori,"||",1)
	s tsub=$p(oeori,"||",2)
	s arcid=$p(^OEORD(ord,"I",tsub,1),"^",2)
	s adm=$p(^OEORD(ord),"^",1)
	
	q:'$d(^DHCWorkLoad(0,"PAADM",adm)) 0
	s wlid=""
	f  s wlid=$o(^DHCWorkLoad(0,"PAADM",adm,wlid)) q:wlid=""  d
	.q:+wlid=0
	.s ordate=$P(^DHCWorkLoad(wlid),"^",5)
	.q:(date '= "")&&(ordate '= date)
	.s arcdr=$P(^DHCWorkLoad(wlid),"^",2)
	.q:arcdr'=arcid
	.s qty=$P(^DHCWorkLoad(wlid),"^",15)
	.s dayct=dayct+qty
	
	q dayct
}
