  ```js
      .replace(/\n/g, '\\n')
      .replace(/\r/g, '\\r')
      .replace(/\t/g, '\\t')
      .replace(/[\b\f]/g, function(c){return c==='\b'?'\\b':'\\f';});
  }
  ```

### SHIM datagrid loadData 重要踩坑
- 现象：在 `loadCov()` 回调里调 `$('#dgCov').datagrid({rowStyler: ...})` 看似"安全"，实际会进入 **init 分支**（`return this.each(...)`），把 `st.data` 重置为 `{total:0, rows:[]}` 并覆盖 `opts.columns`！导致后续 `loadData({total:3, rows:[3]})` 看似成功，但 `getRows()` 返回 0。

- **DEBUG 标记法**：SHIM datagrid 状态拿不到时先 `console.log('DBG st.opts keys=', ...)`，若 keys 只有 `[rowStyler]` 而没有 `columns/pagination`，100% 是第二次 init 覆盖了。

## 部署错误修复：jQuery.fn.dialog 不存在炸整个 JS

### 症状
生产 HISUI 部署后浏览器 console 报错：
```
4VM185 sanyidictmap.js:1108 Uncaught TypeError: $(...).dialog is not a function
```
随后整个页面**没有分页签**（tab 缺失），所有功能异常。

### 根因
- `sanyidictmap.csp` 只通过 `<EXTHEALTH:HEAD>` 注入 HISUI 库，但某些生产现场缺 `jQuery.fn.dialog` 插件（虽然 `<script src="../scripts/sanyidictmap.js">` 之前已 ensure jQuery）。
- `sanyidictmap.js:1108` 原代码：
  ```js
  $dlg.dialog({ title: title, closed: false, modal: true, ... });
  ```
  这行**直接 throw**，由于 `$(function(){...})` 内的 `$('#mainTabs').tabs(...)` 紧随其后，**整个 ready 回调被 abort**，所以主 tabs 不渲染、所有 datagrid 不初始化 → 页面看似"无分页签"。
- 不仅 1108，文件里还有 5 处 `$('#xxx').dialog('close')` 散落在保存成功回调（dlgBind/dlgRelEdit/dlgPlat/dlgDict/dlgProbe），同样会抛。

### 修复（`sanyidictmap.js` line 1097-1188）
- 新增 3 个工具函数：
  - `hasDialog()` — 检测 `$.fn.dialog` 是否可用。
  - `_fallbackOpenDlg(id, title, html, width)` — 自绘 HTML 弹窗（`#id` + `#id-mask`，class `win-fb/win-mask-fb/win-head-fb/win-body-fb/win-btns-fb`，内联 CSS，结构与 HISUI dialog 一致），返回 jQuery 对象。
  - `closeDlg(id)` — `hasDialog()` 走 `$('#'+id).dialog('close')`，否则 `$('#'+id).hide(); $('#'+id+'-mask').remove()`，整段 `try/catch`。
- 重写 `showDlg(id, title, html, width, onOpen, onSave, buttons)`：
  - 有 dialog → 走原 `$dlg.dialog({...})` 路径（外加 try/catch 兜底）。
  - 无 dialog → 走 `_fallbackOpenDlg` + 自绘按钮（`.lb-fb` 绑定到传入的 `buttons` 或默认保存/关闭），按钮 handler 复用 `onSave`/`closeDlg`。
- 全文 5 处 `$('#xxx').dialog('close')` 替换为 `closeDlg('xxx')`：
  - line 699: `dlgBind`
  - line 849: `dlgRelEdit`
  - line 901: `dlgPlat`
  - line 980: `dlgDict`
  - line 1005: `dlgProbe`（在按钮 handler 内）

```js
function hasTabs() {
    return !!(window.jQuery && jQuery.fn && typeof jQuery.fn.tabs === 'function');
}

function setupMainTabs(onSelectCb) {
    var $tabs = $('#mainTabs');
    if (!$tabs.length) return;
    if ($tabs.attr('data-tabs-init') === '1') return;
    var $panels = $tabs.children('div[title]');
    if ($panels.length === 0) return;

    var hisuiOk = false;
    try {
        if (hasTabs()) {
            $tabs.tabs({ onSelect: onSelectCb });
            // 关键: 不光判断函数存在，还要看它真正有没有把面板隐藏。
            // 存在但没生效也要回退。
            var visCnt = 0;
            $panels.each(function () {
                if (this.style && this.style.display !== 'none') { visCnt++; }
            });
            hisuiOk = (visCnt <= 1);
        }
    } catch (eTabs) {
        // 真正的 TypeError 也被静默吞 — 现在走 console.warn 暴露问题
        if (window.console) try { console.warn('[sanyidictmap] $.fn.tabs 调用失败, 启用自绘分页签:', eTabs && eTabs.message); } catch (_e0) {}
        hisuiOk = false;
    }

    if (hisuiOk) { $tabs.attr('data-tabs-init', '1'); return; }

    // 自绘回退: 顶部 tab header, 点击切换 display:block/none
    var $head = $('<div class="mtabs-head" style="display:flex;border-bottom:2px solid #95B8D7;...">');
    $panels.each(function (idx) {
        var $p = $(this);
        var title = $p.attr('title') || ('Tab' + (idx + 1));
        var $t = $('<a class="mtab" data-idx="...">...</a>').text(title);
        $t.on('click', function () {
            $head.find('.mtab').css({ background:'transparent', fontWeight:'normal', color:'#444' });
            $t.css({ background:'#fff', fontWeight:'bold', color:'#1671d4' });
            $panels.css('display', 'none');
            $p.css('display', 'block');
            if (onSelectCb) try { onSelectCb(title, idx); } catch (eSel) {}
        });
        $head.append($t);
    });
    $tabs.prepend($head);
    $head.find('.mtab').first().trigger('click');
    $tabs.attr('data-tabs-init', '1');
    if (window.console) try { console.info('[sanyidictmap] 已切换到自绘分页签模式'); } catch (_e1) {}
}
```

### 关键设计点
1. **判断标准不只是 "$.fn.tabs 存在"，还要 "它真的把面板隐藏了"**。如果存在但不生效（生产 HISUI 偶有这种情况），visCnt=4，hisuiOk=false，自动回退自绘。
2. **被 `try/catch` 吞掉的 throw 现在会打 console.warn**，未来再有 HISUI 组件缺失能从控制台直接看到线索（不用再瞎猜）。
3. **自绘 tab header 内联 CSS**，模仿 HISUI 默认 tabs 的样式（蓝色边框 + 渐变背景 + 激活态白底加粗），不依赖任何外部 CSS。
4. **onSelect 回调签名和 HISUI tabs 一致 `(title, index)`**，onSelect 里那段 dgDict/dgPlat resize 不用改。

### 关键经验（再加一条）
5. **`try/catch` 是双刃剑**：它防"页面全挂"很好，但**完全吞掉 throw 会让调试变噩梦**——生产问题没有任何 console 线索。修复时除了"自绘回退"，还要顺手 `console.warn` 暴露被吞的 throw，让问题能被快速定位。
6. **判断一个 jQuery 插件是否"真的能用"，不能只看 `$.fn.xxx` 是否存在**——还要看它实际效果（返回的 DOM 结构、display 状态等）。HISUI 的插件存在但"半残"的情况比想象中多。
7. **同一类 bug 不会只有一处**。dialog 缺失往往伴随 tabs/combobox/pagination 缺失，因为它们可能来自 HISUI 同一个 `easyui-tabs.js` / `easyui-combo.js` 模块。修一个就要顺手把同类的也加固。
